import assert from 'node:assert/strict'
import fs from 'node:fs'
import path from 'node:path'
import { execFileSync } from 'node:child_process'
import { chromium } from '/Users/caseytalbot/Projects/rangeboard/node_modules/playwright/index.mjs'

// R1 honest-readouts capture: Prepare validity, KPI tiles, Projection, selected-threat readout,
// with and without an active defended asset. Same fixture as R0. Run against an owned port only.
const [,, fixture, url, outDir, repo] = process.argv
assert.ok(fixture && url && outDir && repo, 'usage: fixture url outDir repo')
assert.notEqual(new URL(url).port, '5174', 'never the user dev server')
fs.mkdirSync(outDir, { recursive: true })
const raw = fs.readFileSync(fixture, 'utf8')
const result = {
  status: 'pending', date: '2026-09-07',
  commit: execFileSync('git', ['rev-parse', 'HEAD'], { cwd: repo, encoding: 'utf8' }).trim(),
  url, viewport: { width: 1400, height: 900 }, panelWidth: 320, captures: [], pageErrors: [], consoleErrors: [],
}
let browser, ctx
try {
  browser = await chromium.launch({ headless: true })
  ctx = await browser.newContext({ viewport: result.viewport })
  await ctx.addInitScript(() => localStorage.setItem('rangeboard.rightpanel.width', '320'))
  const page = await ctx.newPage()
  page.on('pageerror', e => result.pageErrors.push(e.message))
  page.on('console', m => { if (m.type() === 'error') result.consoleErrors.push(m.text()) })
  await page.goto(url, { waitUntil: 'load' })
  await page.waitForFunction(() => window.__store && document.querySelector('.konvajs-content canvas'))
  await page.evaluate(async raw => {
    const { parseAutosave } = await import('/src/io/persist.ts')
    window.__store.getState().loadScenario(parseAutosave(raw))
  }, raw)
  const ids = await page.evaluate(() => {
    const s = window.__store.getState().scenario
    return { threat: s.entities.find(e => e.kind === 'uas').id, asset: s.entities.find(e => e.kind === 'asset').id }
  })
  result.ids = ids
  const events = await page.evaluate(async () => {
    const { engagementPicture } = await import('/src/analysis/impact.ts')
    return engagementPicture(window.__store.getState().scenario).report.events
  })
  const endS = Math.max(...events.map(e => e.tEffectS)) + 1
  result.expectedKills = await page.evaluate(async () => {
    const { engagementPicture } = await import('/src/analysis/impact.ts')
    return engagementPicture(window.__store.getState().scenario).report.totals.expectedKills
  })
  const textOf = sel => page.locator(sel).allTextContents()
  const titles = sel => page.locator(sel).evaluateAll(n => n.map(el => ({ text: el.textContent, title: el.getAttribute('title'), aria: el.getAttribute('aria-label') })))
  async function shot(name, extra = {}) {
    await page.mouse.move(10, 10)
    await page.waitForTimeout(250)
    await page.screenshot({ path: path.join(outDir, `${name}.png`), fullPage: true })
    result.captures.push({ name, playback: await page.evaluate(() => window.__store.getState().playback),
      rightPanelText: await textOf('.right-panel'), ...extra })
  }
  async function prepare(name) {
    await page.click('#board-phase-prepare')
    await page.evaluate(() => { window.__store.getState().clearSelection(); window.__store.getState().setRightTab('analysis') })
    await page.waitForSelector('.board-preparation')
    for (const d of await page.locator('.board-preparation details').all()) await d.evaluate(el => { el.open = true })
    await shot(name, { validity: await textOf('.board-preparation'), buttons: await titles('.board-preparation button') })
  }
  async function execute(name, scrubS) {
    await page.click('#board-phase-execute')
    await page.evaluate(() => { window.__store.getState().clearSelection(); window.__store.getState().setRightTab('analysis') })
    await page.waitForSelector('.right-panel [data-widget-type]')
    if (scrubS != null) await page.evaluate(t => window.__store.getState().scrubTo(t), scrubS)
    await page.waitForTimeout(200)
    const widgets = await page.locator('.right-panel [data-widget-type]').evaluateAll(n => n.map(el => ({ type: el.dataset.widgetType, text: el.textContent })))
    const kpi = await titles('[data-widget-type="kpi-tiles"] .tile')
    await page.evaluate(id => window.__store.getState().select({ entityIds: [id] }), ids.threat)
    await page.waitForSelector('.entity-workspace')
    await page.click('#entity-section-telemetry')
    await page.waitForTimeout(200)
    const selected = { compact: await textOf('.readout.compact'), telemetry: await textOf('#entity-workspace-content') }
    await page.locator('.entity-workspace-header button').click()
    await page.waitForSelector('.right-panel [data-widget-type]')
    await shot(name, { widgets, kpi, selected })
  }
  await prepare('prepare-with-asset')
  await execute('execute-ready', null)
  await execute('execute-scrub-end', endS)
  await page.evaluate(id => window.__store.getState().updateEntity(id, { enabled: false }), ids.asset)
  await prepare('prepare-no-asset')
  await execute('execute-no-asset-scrub-end', endS)
  await page.evaluate(id => window.__store.getState().updateEntity(id, { enabled: true }), ids.asset)
  result.status = 'captured'
} catch (e) {
  result.status = 'failed'; result.error = String(e); process.exitCode = 1
} finally {
  await ctx?.close(); await browser?.close()
  fs.writeFileSync(path.join(outDir, 'r1-capture.json'), JSON.stringify(result, null, 2) + '\n')
}
