// Right panel R3 browser check: the placement preview on the board. Evidence of a dated
// run, not a test. Extends the R2 script (right-panel-r2-capture.mjs.txt): it adds a range
// boundary to the fixture, runs one placement optimization, and captures the preview on
// and off and each clear.
// Usage: node right-panel-r3-capture.mjs <base-url> <out-dir> <repo-root>
import { chromium } from 'playwright'
import { mkdirSync, readFileSync, writeFileSync } from 'node:fs'
import { join } from 'node:path'

const [BASE_URL = 'http://localhost:5291', OUT = './out', ROOT = '.'] = process.argv.slice(2)
const FIXTURE = 'docs/field-te/evidence/onboarding-2026-09-04/firing-scenario.json'
const RAW = readFileSync(join(ROOT, FIXTURE), 'utf8')
const VIEWPORT = { width: 1400, height: 900 }
const WIDTHS = [408, 280]
const THEMES = { night: 'dark', paper: 'light' }
// The fixture has no range boundary; the optimizer needs one.
const RANGE = {
  id: 'bnd-r3-range',
  name: 'R3 range',
  type: 'range',
  closed: true,
  altBandFt: null,
  points: [
    { x: -2500, y: -2500 },
    { x: 2500, y: -2500 },
    { x: 2500, y: 2500 },
    { x: -2500, y: 2500 },
  ],
}
// The run goes through the screen: the default goal (Minimize leakers), the effector and
// the first radar checked, and a small budget to keep it short.
const SPEC = {
  move: ['Coyote Block 2 interceptor 1', 'Giraffe 1X 3D radar (Saab) 1'],
  evaluations: '6',
  replicates: '4',
}
mkdirSync(join(OUT, 'all'), { recursive: true })

const browser = await chromium.launch()
const st = (page, fn, arg) => page.evaluate(fn, arg)
const waitFor = (page, fn, arg, timeout = 180000) =>
  page.waitForFunction(fn, arg, { timeout, polling: 100 })
const settle = (page) =>
  page.evaluate(
    () => new Promise((r) => requestAnimationFrame(() => requestAnimationFrame(() => setTimeout(r, 150)))),
  )

async function open(width, theme) {
  const ctx = await browser.newContext({ viewport: VIEWPORT })
  await ctx.addInitScript((w) => localStorage.setItem('rangeboard.rightpanel.width', String(w)), width)
  const page = await ctx.newPage()
  const errors = []
  page.on('pageerror', (e) => errors.push({ kind: 'pageerror', text: e.message.slice(0, 300) }))
  page.on('console', (m) => {
    if (m.type() === 'error') errors.push({ kind: 'console', text: m.text().slice(0, 300) })
  })
  page.on('dialog', (d) => d.accept())
  await page.goto(BASE_URL, { waitUntil: 'load' })
  await waitFor(page, () => document.querySelector('.konvajs-content canvas') && window.__store, null, 60000)
  await st(page, (t) => window.__store.getState().setTheme(t), theme)
  await st(page, async ({ raw, range }) => {
    const { parseAutosave } = await import('/src/io/persist.ts')
    const s = window.__store.getState()
    s.loadScenario(parseAutosave(raw))
    window.__store.getState().patchScenario({ boundaries: [range] })
  }, { raw: RAW, range: RANGE })
  await settle(page)
  return { ctx, page, errors }
}

async function go(page, phase, screen, section = null) {
  await st(page, (r) => window.__store.getState().requestPanel(r), { phase, screen, section })
  await settle(page)
}

// The screen keeps its picks in local state, so a remount after a phase change needs them again.
async function runPlacement(page) {
  for (const name of SPEC.move) await page.check(`#insp-tabpanel input[aria-label="Move ${name}"]`)
  const budget = page.locator('#insp-tabpanel input[type="number"]')
  await budget.nth(0).fill(SPEC.evaluations)
  await budget.nth(1).fill(SPEC.replicates)
  await page.click('#insp-tabpanel button:text-is("Run placement")')
  await waitFor(page, () => ['done', 'error'].includes(window.__store.getState().mc.optimize.status))
  await settle(page)
  return st(page, () => {
    const o = window.__store.getState().mc.optimize
    const ranks = o.summary?.candidates ?? []
    // The first candidate that moves from the start shows a leader; the start shows none.
    const moved = ranks.findIndex((c) => c.evaluationIndex !== 0) + 1
    return { status: o.status, error: o.error, candidates: ranks.length, moved, objective: o.summary?.objective }
  })
}

// What the board draws for the preview, read from the Konva stage: the dashed leaders
// and the outline frames at 0.7 opacity. The store field is read beside it.
function boardPreview() {
  const stage = window.Konva?.stages?.[0]
  const s = window.__store.getState()
  if (!stage) return { stage: false, field: s.ui.placementPreview?.rank ?? null }
  const leaders = stage
    .find('Line')
    .filter((l) => JSON.stringify(l.dash()) === '[4,4]' && l.points().length === 4 && !l.strokeScaleEnabled())
  const frames = stage
    .find('Path')
    .filter((p) => p.getParent()?.opacity() === 0.7 && !p.fill())
  return { stage: true, field: s.ui.placementPreview?.rank ?? null, leaders: leaders.length, frames: frames.length }
}

// Show the Candidates section; the board stays on the left.
async function showCandidates(page) {
  await st(page, () => {
    const rows = document.querySelector('#insp-tabpanel .r2b-cands')
    rows?.scrollIntoView({ block: 'center' })
  })
  await settle(page)
}

// Move the pointer off any control and wait out any tooltip, so Escape reaches the board.
async function closeTips(page) {
  await page.mouse.move(VIEWPORT.width / 2, 40)
  await page.waitForTimeout(300)
  await settle(page)
}

// The three pass criteria read in the page (unchanged from the R2 script).
function measure() {
  const panel = document.getElementById('insp-tabpanel')
  if (!panel) return { panel: false }
  const pr = panel.getBoundingClientRect()
  const tab = document.querySelector('.insp-tabs [role="tab"][aria-selected="true"]')?.textContent?.trim()
  const hidden = (el, cs, r) =>
    r.width <= 1 ||
    r.height <= 1 ||
    cs.visibility === 'hidden' ||
    cs.display === 'none' ||
    /rect\(0/.test(cs.clip) ||
    /inset\(50%/.test(cs.clipPath)
  const path = (el) => {
    const parts = []
    for (let n = el; n && n !== panel && parts.length < 4; n = n.parentElement) {
      const c = typeof n.className === 'string' && n.className ? '.' + n.className.trim().split(/\s+/).join('.') : ''
      parts.unshift(n.tagName.toLowerCase() + (n.id ? '#' + n.id : '') + c)
    }
    return parts.join(' > ')
  }
  const rowOf = (el) => el.closest('tr, li, [role="row"], summary, button, label, .insp-row, [class*="row"], .field, .pstat')
  const titled = (el) => {
    const row = rowOf(el)
    for (let n = el; n && n !== panel; n = n.parentElement) {
      if (n.hasAttribute('title') && n.getAttribute('title')) return true
      if (n === row) break
    }
    return false
  }
  const wide = []
  const innerScroll = []
  const clipped = []
  for (const el of panel.querySelectorAll('*')) {
    const r = el.getBoundingClientRect()
    const cs = getComputedStyle(el)
    if (hidden(el, cs, r)) continue
    if (r.width > panel.clientWidth + 0.5)
      wide.push({ el: path(el), width: Math.round(r.width * 10) / 10 })
    if (/(auto|scroll)/.test(cs.overflowX) && el.scrollWidth > el.clientWidth + 1)
      innerScroll.push({ el: path(el), scrollWidth: el.scrollWidth, clientWidth: el.clientWidth })
    if (el.closest('svg') || el.tagName === 'CANVAS') continue
    const text = (el.textContent || '').trim()
    if (!text) continue
    const ownText = [...el.childNodes].some((n) => n.nodeType === 3 && n.textContent.trim())
    const clips = /(hidden|clip)/.test(cs.overflowX) || cs.textOverflow === 'ellipsis'
    if (!clips || !(ownText || cs.textOverflow === 'ellipsis')) continue
    // R3: a title on a child span is recorded for the reading by hand.
    if (el.scrollWidth > el.clientWidth + 1 && !titled(el))
      clipped.push({
        el: path(el),
        text: text.slice(0, 60),
        scrollWidth: el.scrollWidth,
        clientWidth: el.clientWidth,
        childTitle: el.querySelector('[title]')?.getAttribute('title') ?? null,
      })
  }
  // R3: the candidate action buttons must show their whole label on one line.
  const acts = [...panel.querySelectorAll('.r2b-cand-acts button')].map((b) => {
    const r = b.getBoundingClientRect()
    const cs = getComputedStyle(b)
    return {
      label: b.getAttribute('aria-label'),
      pressed: b.getAttribute('aria-pressed'),
      fits: b.scrollWidth <= b.clientWidth + 1 && r.height <= parseFloat(cs.minHeight) + 1,
      width: Math.round(r.width),
      height: Math.round(r.height),
      background: cs.backgroundColor,
    }
  })
  return {
    panel: true,
    tab,
    panelWidth: Math.round(pr.width),
    scrollWidth: panel.scrollWidth,
    clientWidth: panel.clientWidth,
    sidewaysScroll: panel.scrollWidth > panel.clientWidth || wide.length > 0,
    wide: wide.slice(0, 10),
    innerScroll: innerScroll.slice(0, 10),
    clipped: clipped.slice(0, 20),
    clippedCount: clipped.length,
    acts,
  }
}

const captures = []
async function record(ctxInfo, state, width, themeName, extra = {}) {
  const { page, errors } = ctxInfo
  const m = await page.evaluate(measure)
  const board = await page.evaluate(boardPreview)
  const name = `placement-${state}-${width}-${themeName}.png`
  await page.screenshot({ path: join(OUT, 'all', name) })
  const errs = errors.splice(0)
  const row = {
    file: name,
    state,
    width,
    theme: themeName,
    errors: errs,
    board,
    ...m,
    ...extra,
  }
  // The R2 rule counts a clip whose title sits on a child span as a fail; those are read by
  // hand, and the pass takes a clip whose child title holds the whole text.
  row.clippedNoTitle = (m.clipped ?? []).filter((x) => !(x.childTitle && x.childTitle.startsWith(x.text)))
  row.pass =
    errs.length === 0 &&
    m.panel &&
    !m.sidewaysScroll &&
    row.clippedNoTitle.length === 0 &&
    (m.acts ?? []).every((a) => a.fits)
  captures.push(row)
  console.log(
    `${row.pass ? 'pass' : 'FAIL'} ${name} tab=${m.tab} sw=${m.scrollWidth}/${m.clientWidth} wide=${m.wide?.length} clipped=${m.clippedCount} errs=${errs.length} board=${JSON.stringify(board)}`,
  )
  return row
}

const clickPreview = async (page, rank) => {
  await page.click(`#insp-tabpanel button[aria-label="Preview placement ${rank}"]`)
  await settle(page)
}

const behavior = {}
for (const width of WIDTHS) {
  for (const [themeName, theme] of Object.entries(THEMES)) {
    const c = await open(width, theme)
    const { page } = c
    await go(page, 'prepare', 'placement')
    const run = await runPlacement(page)
    await showCandidates(page)
    const key = `${width}-${themeName}`
    const b = { run }
    await closeTips(page)
    b.off = (await record(c, 'off', width, themeName)).board
    await clickPreview(page, run.moved)
    await closeTips(page)
    b.on = (await record(c, 'on', width, themeName)).board

    // Clear 1: Escape on the board, after any tooltip has closed.
    await closeTips(page)
    await page.keyboard.press('Escape')
    await settle(page)
    b.escape = (await record(c, 'escape', width, themeName)).board

    // Clear 2: a phase change away from Prepare, then back.
    await clickPreview(page, run.moved)
    b.on2 = await st(page, boardPreview)
    await go(page, 'execute', 'now')
    b.phase = await st(page, boardPreview)
    await record(c, 'phase-execute', width, themeName)
    await go(page, 'prepare', 'placement')
    await showCandidates(page)
    b.phaseBack = await st(page, boardPreview)

    // A rerun on the same board hides the old preview with no clear call.
    await clickPreview(page, run.moved)
    b.beforeRerun = await st(page, boardPreview)
    b.rerun = await runPlacement(page)
    b.afterRerun = await st(page, boardPreview)
    await showCandidates(page)
    await closeTips(page)
    await record(c, 'rerun', width, themeName)

    // Clear 3: Apply installs a copy and drops the preview.
    const moved = b.rerun.moved
    await clickPreview(page, moved)
    b.beforeApply = await st(page, boardPreview)
    const name = await st(page, () => window.__store.getState().scenario.name)
    await page.click(`#insp-tabpanel button[aria-label="Apply placement ${moved}"]`)
    await settle(page)
    b.apply = await st(page, boardPreview)
    b.appliedName = await st(page, () => window.__store.getState().scenario.name)
    b.applyRenamed = b.appliedName === `${name} (placement ${moved})`
    await closeTips(page)
    await record(c, 'apply', width, themeName)

    const shown = (x) => x.field != null && x.leaders > 0 && x.frames > 0
    // The rerun keeps the old field: the board hides it because its result is no longer current.
    const gone = (x) => x.leaders === 0 && x.frames === 0
    b.pass =
      run.status === 'done' &&
      gone(b.off) &&
      shown(b.on) &&
      b.escape.field == null && gone(b.escape) &&
      shown(b.on2) &&
      b.phase.field == null && gone(b.phase) &&
      b.phaseBack.field == null && gone(b.phaseBack) &&
      shown(b.beforeRerun) &&
      b.rerun.status === 'done' && b.afterRerun.field === run.moved && gone(b.afterRerun) &&
      shown(b.beforeApply) &&
      b.apply.field == null && gone(b.apply) && b.applyRenamed
    behavior[key] = b
    console.log(key, JSON.stringify(b))
    await c.ctx.close()
  }
}

await browser.close()
const summary = {
  captures: captures.length,
  passed: captures.filter((r) => r.pass).length,
  failed: captures.filter((r) => !r.pass).map((r) => r.file),
  behavior: Object.fromEntries(Object.entries(behavior).map(([k, v]) => [k, v.pass])),
}
writeFileSync(
  join(OUT, 'findings.json'),
  JSON.stringify({ baseUrl: BASE_URL, viewport: VIEWPORT, fixture: FIXTURE, range: RANGE, spec: SPEC, summary, behavior, captures }, null, 2) + '\n',
)
console.log(JSON.stringify(summary))
