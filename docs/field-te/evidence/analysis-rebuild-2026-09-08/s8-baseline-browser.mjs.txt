// S8 browser check: load, run MC, pin, reload, compare, storage listing, clear.
import assert from 'node:assert/strict'
import { readFileSync } from 'node:fs'
import { chromium } from '/Users/caseytalbot/Projects/rangeboard-wt/s8/node_modules/playwright/index.mjs'

const BASE = 'http://localhost:5180'
const ROOT = '/Users/caseytalbot/Projects/rangeboard-wt/s8'
const OUT = '/private/tmp/claude-501/-Users-caseytalbot/aa5d92d8-5644-48c9-aa4a-8d379b605167/scratchpad'
const RAW = readFileSync(`${ROOT}/docs/field-te/evidence/onboarding-2026-09-04/firing-scenario.json`, 'utf8')
const KEY = 'rangeboard.baseline.v1'

const browser = await chromium.launch({ headless: true })
const ctx = await browser.newContext({ viewport: { width: 1265, height: 900 } })
await ctx.addInitScript(() => localStorage.setItem('rangeboard.rightpanel.width', '320'))
const page = await ctx.newPage()
const errors = []
page.on('pageerror', (e) => errors.push(e.message))
page.on('console', (m) => { if (m.type() === 'error') errors.push(m.text().slice(0, 200)) })
const waitFor = (fn, arg, timeout = 60000) => page.waitForFunction(fn, arg, { timeout })
const state = (path) => page.evaluate((p) => p.split('.').reduce((o, k) => o?.[k], window.__store.getState()), path)
const results = {}

async function open() {
  await page.goto(BASE, { waitUntil: 'load' })
  await waitFor(() => document.querySelector('.konvajs-content canvas') && window.__store)
}
async function runMc() {
  await page.evaluate(() => window.__store.getState().runMonteCarlo())
  await waitFor(() => ['done', 'error'].includes(window.__store.getState().mc.status), null, 120000)
  assert.equal(await state('mc.status'), 'done')
}
async function completeTrial() {
  await page.click('#board-phase-analyze')
  await page.evaluate(() => window.__store.getState().startRun())
  await waitFor(() => ['ready', 'error'].includes(window.__store.getState().trial.status), null, 120000)
  assert.equal(await state('trial.status'), 'ready')
  await page.evaluate(async () => {
    const s = window.__store.getState()
    const { durationOf } = await import('/src/state/selectors.ts')
    s.completeRun(durationOf(s.scenario))
  })
}
async function openCompare() {
  await page.click('#board-phase-analyze')
  await page.click('nav.lens button:has-text("Compare")')
  await waitFor(() => document.querySelector('.insp-row[aria-label="Baseline"]'))
}
const baselineRow = () => page.evaluate(() => {
  const row = document.querySelector('.insp-row[aria-label="Baseline"]')
  return { text: row.textContent, buttons: [...row.querySelectorAll('button')].map(b => ({ text: b.textContent, disabled: b.disabled })) }
})

// 1. Load the firing fixture through the real parser; run the ensemble.
await open()
await page.evaluate(async (raw) => {
  const { parseAutosave } = await import('/src/io/persist.ts')
  window.__store.getState().loadScenario(parseAutosave(raw))
}, RAW)
results.threats = await page.evaluate(() => window.__store.getState().scenario.entities.filter(e => e.kind === 'uas').length)
assert.ok(results.threats > 0, 'fixture has threats')
await completeTrial()
await runMc()
results.replicates = await state('mc.summary.replicates')

// 2. Compare before a pin: Pin enabled, no baseline, five columns.
await openCompare()
results.headersBefore = await page.evaluate(() => [...document.querySelectorAll('section[aria-label="Compare"] thead th')].map(th => th.textContent))
results.beforePin = await baselineRow()
assert.match(results.beforePin.text, /No pinned baseline/)
assert.deepEqual(results.beforePin.buttons, [{ text: 'Pin', disabled: false }])

// 3. Pin, then wait for the debounced write.
await page.click('.insp-row[aria-label="Baseline"] button:has-text("Pin")')
await waitFor((key) => localStorage.getItem(key) !== null, KEY)
results.pinned = await page.evaluate(() => {
  const b = window.__store.getState().baseline
  return { label: b.label, pinnedAt: b.pinnedAt, hash: b.scenarioHash.length, seed: b.mc.seed, replicates: b.mc.replicates, name: b.scenario.name }
})
results.afterPin = await baselineRow()
assert.match(results.afterPin.text, /Baseline · /)
assert.deepEqual(results.afterPin.buttons.map(b => b.text), ['Clear', 'Pin'])
results.savedBytes = await page.evaluate((key) => localStorage.getItem(key).length, KEY)
results.savedNullCount = await page.evaluate((key) => (localStorage.getItem(key).match(/:null/g) || []).length, KEY)

// 4. Reload: the baseline survives and no recovery copy appears.
await page.reload({ waitUntil: 'load' })
await waitFor(() => document.querySelector('.konvajs-content canvas') && window.__store)
results.afterReload = await page.evaluate(() => {
  const b = window.__store.getState().baseline
  return { label: b?.label, pinnedAt: b?.pinnedAt, nanRestored: Number.isNaN(b?.mc.firstDetectS.p10) || Number.isFinite(b?.mc.firstDetectS.p10),
    keys: Object.keys(localStorage).filter(k => k.startsWith('rangeboard.')).sort(), entities: window.__store.getState().scenario.entities.length }
})
assert.equal(results.afterReload.label, results.pinned.label)
assert.equal(results.afterReload.pinnedAt, results.pinned.pinnedAt)
assert.ok(!results.afterReload.keys.some(k => k.includes('.recovered.')), 'no recovery key')

// 5. Storage listing and the Storage panel name the key.
results.listing = await page.evaluate(async () => {
  const { readStorageUsage } = await import('/src/io/storageManagement.ts')
  return readStorageUsage().entries.filter(e => e.key.includes('baseline'))
})
assert.ok(results.listing.some(e => e.key === KEY && e.kind === 'live' && e.bytes > 0), 'listing has the baseline key')
await page.click('button[aria-controls="storage-panel"]')
await waitFor(() => document.getElementById('storage-panel')?.textContent.includes('rangeboard.baseline.v1'))
results.storagePanelShowsKey = true
await page.keyboard.press('Escape')

// 6. A completed trial and a fresh ensemble: Compare shows the Baseline column and the dashed mean.
await completeTrial()
await runMc()
await openCompare()
await waitFor(() => document.querySelector('section[aria-label="Compare"] table'))
results.compare = await page.evaluate(() => {
  const section = document.querySelector('section[aria-label="Compare"]')
  const headers = [...section.querySelectorAll('thead th')].map(th => th.textContent)
  const rows = [...section.querySelectorAll('tbody tr')].map(tr => [...tr.children].map(c => c.textContent))
  const legend = [...section.querySelectorAll('.chart-legend span')].map(s => s.textContent)
  const dashed = section.querySelector('.recharts-line path[stroke-dasharray]') !== null
  const img = section.querySelector('[role="img"]').getAttribute('aria-label')
  const details = section.querySelector('details.analysis-disclosure').textContent
  const panel = section.closest('.right-panel, aside, [class*="panel"]')
  return { headers, rows, legend, dashed, img, details: details.slice(0, 400), width: section.getBoundingClientRect().width,
    overflow: section.scrollWidth > section.clientWidth, panelClass: panel?.className }
})
assert.deepEqual(results.compare.headers, ['Quantity', 'Trial', 'Ensemble mean', '95 % band', 'Baseline', 'Current − Baseline', 'Flag'])
assert.ok(results.compare.legend.some(l => l.includes('Baseline mean')), 'legend has the baseline')
assert.ok(results.compare.dashed, 'dashed baseline line rendered')
await page.screenshot({ path: `${OUT}/s8-compare-pinned.png`, fullPage: false })

// 7. Clear: written as null, survives reload as no baseline.
await page.click('.insp-row[aria-label="Baseline"] button:has-text("Clear")')
await waitFor((key) => localStorage.getItem(key) === '{"schemaVersion":1,"baseline":null}', KEY)
results.afterClear = await baselineRow()
assert.match(results.afterClear.text, /No pinned baseline/)
await page.reload({ waitUntil: 'load' })
await waitFor(() => document.querySelector('.konvajs-content canvas') && window.__store)
results.afterClearReload = await page.evaluate(() => ({ baseline: window.__store.getState().baseline,
  keys: Object.keys(localStorage).filter(k => k.startsWith('rangeboard.')).sort() }))
assert.equal(results.afterClearReload.baseline, null)
assert.ok(!results.afterClearReload.keys.some(k => k.includes('.recovered.')), 'no recovery key after clear')

// 8. Save-shape guard: a baseline save with a corrupt summary is copied to recovery and not loaded.
await page.evaluate((key) => {
  const save = JSON.parse(localStorage.getItem(key))
  localStorage.setItem(key, JSON.stringify({ schemaVersion: 1, baseline: { pinnedAt: 'x', label: 'bad', scenarioHash: 'h', scenario: {}, mc: {} } }))
  return save
}, KEY)
await page.reload({ waitUntil: 'load' })
await waitFor(() => document.querySelector('.konvajs-content canvas') && window.__store)
results.corrupt = await page.evaluate(async () => {
  const { getRecoveryNotices } = await import('/src/io/recovery.ts')
  return { baseline: window.__store.getState().baseline, notices: getRecoveryNotices().map(n => n.message),
    recovered: Object.keys(localStorage).filter(k => k.startsWith('rangeboard.baseline.recovered.')) }
})
assert.equal(results.corrupt.baseline, null)
assert.equal(results.corrupt.recovered.length, 1)
assert.match(results.corrupt.notices.join(' '), /saved baseline could not be read/)
results.storageButton = await page.textContent('button[aria-controls="storage-panel"]')

results.errors = errors
console.log(JSON.stringify(results, null, 2))
await ctx.close()
await browser.close()
