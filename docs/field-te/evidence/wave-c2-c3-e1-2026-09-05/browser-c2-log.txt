    }))

    if (metrics.find(x => x.label === 'Defeat')?.value !== expectedDefeat) throw Error('Defeat differs from MC mean')

    if (!metrics.find(x => x.label === 'Late E[k]')?.sub.includes('envelope')) throw Error('Late kills missing envelope basis')
    return { status: s.mc.status, expectedDefeat, meanKills: mc.kills.mean.at(-1), meanLeakers: mc.leakers.mean.at(-1), deterministicKills: p.report.totals.expectedKills, deterministicEvents: p.report.events.length, metrics, gates: document.querySelector('.requirements-block')?.textContent }

  await download.saveAs('output/playwright/wave-run-export.json')
  await page.evaluate(() => { const s = window.__store.getState(); s.scrubTo(100); })
  await page.locator('.recharts-wrapper').first().waitFor()

    typeFloor: [...document.querySelectorAll('.right-panel *')].filter(el=>[...el.childNodes].some(n=>n.nodeType===3&&n.textContent.trim()) && parseFloat(getComputedStyle(el).fontSize)<10 && el.tagName!=='text').map(el=>({cls:el.className,text:el.textContent}))
  }))
  if (!charts.paths || !charts.bandPaths) throw Error('MC chart line or quantile bands not rendered')

  if (stale.status!=='stale') throw Error('C1 edit did not stale MC result')

  if (restored.kills!==current.meanKills || restored.leakers!==current.meanLeakers) throw Error('Seeded MC changed after undo/recompute')
  await page.evaluate(result=>{window.__waveResults=result},{current,charts,stale,restored,errors})
}
