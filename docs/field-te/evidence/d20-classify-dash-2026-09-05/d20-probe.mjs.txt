// D-20 browser check: under the Full preset a Pd-modeled radar with a classify range
// draws the Pd 0.9 contour and the classify ring with different dashes. Detection
// fixture through the store, one RPS-42 added, the Mavic 3 selected, real Konva nodes read.
import { createRequire } from 'node:module'
const { chromium } = createRequire('/Users/caseytalbot/Projects/rangeboard/package.json')('playwright')
import { writeFileSync } from 'node:fs'
const [baseUrl = 'http://localhost:5361/', outDir = '.'] = process.argv.slice(2)
const browser = await chromium.launch({ headless: true })
const context = await browser.newContext({ viewport: { width: 1400, height: 900 }, deviceScaleFactor: 2 })
await context.addInitScript(() => localStorage.setItem('rangeboard.rightpanel.width', '320'))
const page = await context.newPage()
const errors = []
page.on('pageerror', (e) => errors.push(String(e)))
page.on('console', (m) => { if (m.type() === 'error') errors.push(m.text()) })
await page.goto(baseUrl, { waitUntil: 'networkidle' })
await page.waitForFunction(() => Boolean(window.__store))
const result = await page.evaluate(async () => {
  const { parseAutosave } = await import('/src/io/persist.ts')
  const { fitCameraTo } = await import('/src/state/camera.ts')
  const { RING_STYLE, PD_HIGH_DASH, CLASSIFY_DASH } = await import('/src/canvas/layers/footprintStyle.ts')
  const raw = await (await fetch('/src/playback/__fixtures__/detection-outside-rings.json')).text()
  const s = window.__store.getState()
  s.loadScenario(parseAutosave(raw))
  s.addEntityFromModel('lib-rps42', { x: -4000, y: 3000 })
  s.applyLayerPreset('full')
  const st = window.__store.getState()
  const mavic = st.scenario.entities.find((e) => e.kind === 'uas')
  const rps = st.scenario.entities.find((e) => e.specId === 'lib-rps42')
  st.select({ entityIds: [mavic.id] })
  st.setRightTab('inspect')
  const cam = fitCameraTo([{ x: -10000, y: -3000 }, { x: 2000, y: 9000 }])
  if (cam) window.__store.getState().setCamera(cam)
  return {
    rps: rps?.id, layers: window.__store.getState().ui.layers,
    classifyDash: RING_STYLE.classifyRings.dash, pdHighDash: PD_HIGH_DASH, constant: CLASSIFY_DASH,
  }
})
console.log(JSON.stringify(result))
await page.waitForTimeout(600)
await page.screenshot({ path: `${outDir}/full-preset-rps42-mavic3.png`, animations: 'disabled', timeout: 60000 })
console.log('shot1')
// Read the live Konva nodes: every dashed arc or circle on the stage with its dash.
const nodes = await page.evaluate(async () => {
  const Konva = (await import('/@id/konva')).default ?? (await import('/@id/konva'))
  const stage = Konva.stages?.[0]
  if (!stage) return { error: 'no stage' }
  const shapes = stage.find('Arc, Circle, Ring, Shape')
  const dashed = shapes.filter((n) => Array.isArray(n.dash?.()) && n.dash().length && n.isVisible())
  const counts = {}
  for (const n of dashed) {
    const k = JSON.stringify(n.dash())
    counts[k] = (counts[k] ?? 0) + 1
  }
  return { dashedShapes: dashed.length, byDash: counts }
})
console.log(JSON.stringify(nodes))
// Zoomed on the RPS-42 so the two dashes are readable.
await page.evaluate(async () => {
  const { fitCameraTo } = await import('/src/state/camera.ts')
  const cam = fitCameraTo([{ x: -9500, y: -2500 }, { x: 1500, y: 8500 }])
  if (cam) window.__store.getState().setCamera({ ...cam, scale: cam.scale * 1.6 })
})
await page.waitForTimeout(400)
await page.screenshot({ path: `${outDir}/full-preset-rps42-zoom.png`, animations: 'disabled', timeout: 60000 })
const out = { baseUrl, result, nodes, errors }
writeFileSync(`${outDir}/browser-results.json`, JSON.stringify(out, null, 2))
console.log(JSON.stringify(out, null, 2))
await browser.close()
