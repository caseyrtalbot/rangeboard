// sRGB relative luminance and the WCAG contrast ratio. Every token is opaque
// after the fold, so a colour needs no compositing; the helper keeps the alpha
// path anyway, to prove that it is never taken.
const CONTRAST_FN = `
  const srgb = (c) => { const v = c / 255; return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4) }
  const lum = ([r, g, b]) => 0.2126 * srgb(r) + 0.7152 * srgb(g) + 0.0722 * srgb(b)
  const parse = (s) => { const m = s.match(/[\\d.]+/g).map(Number); return [m[0], m[1], m[2], m.length > 3 ? m[3] : 1] }
  const over = (fg, bg) => fg.slice(0, 3).map((c, i) => Math.round(c * fg[3] + bg[i] * (1 - fg[3])))
  const hex = (c) => '#' + c.map((v) => v.toString(16).padStart(2, '0')).join('')
  const ratio = (a, b) => { const l1 = lum(a), l2 = lum(b); return (Math.max(l1, l2) + 0.05) / (Math.min(l1, l2) + 0.05) }
`

async function loadFixture(page) {
  await page.evaluate(async () => {
    const { parseAutosave } = await import('/src/io/persist.ts')
    const raw = await (await fetch('/docs/field-te/evidence/onboarding-2026-09-04/firing-scenario.json')).text()
    const s = window.__store.getState()
    s.loadScenario(parseAutosave(raw))
    s.setRightTab('analysis')
    s.applyLens('run-summary')
    s.startRun()
    s.advanceRun(300)
  })
  await page.waitForTimeout(400)
}

async function openTab(page, tab) {
  await page.evaluate((t) => {
    const s = window.__store.getState()
    s.setRightTab(t)
    if (t === 'inspect') {
      const e = s.scenario.entities.find((x) => x.kind === 'system') ?? s.scenario.entities[0]
      if (e) s.select({ entityIds: [e.id] })
    }
    if (t === 'library') {
      const spec = s.scenario.library[0]
      if (spec) s.select({ libraryId: spec.id })
    }
  }, tab)
  await page.waitForTimeout(350)
}

const shot = async (page, name) => {
  await page.screenshot({ path: `${OUT}/${name}.png` })
  say(`screenshot ${OUT}/${name}.png`)
}

// 1. The tokens as the browser resolves them, and the board grain.
const tokens = await page.evaluate(() => {
  const cs = getComputedStyle(document.documentElement)
  const names = [
    '--surface-board', '--surface-panel', '--surface-sunken', '--surface-hover',
    '--rule', '--rule-strong', '--text-primary', '--text-secondary', '--text-tertiary',
    '--text-disabled', '--ink', '--status-nominal', '--status-warning', '--status-critical',
    '--radius-control', '--dur-fast', '--dur', '--accent', '--border-focus',
  ]
  const host = document.querySelector('.canvas-host')
  const hcs = host ? getComputedStyle(host) : null
  return {
    values: Object.fromEntries(names.map((n) => [n, cs.getPropertyValue(n).trim()])),
    boardBackground: hcs ? hcs.backgroundColor : null,
    boardGrain: hcs ? hcs.backgroundImage.slice(0, 60) + (hcs.backgroundImage.length > 60 ? '…' : '') : null,
    grainDeclared: cs.getPropertyValue('--board-grain').includes('feTurbulence'),
  }
})
say('tokens', JSON.stringify(tokens))

// 3. Contrast: the three text steps and the three state inks against the panel
//    surface. A token is resolved through a probe element so the browser hands
//    back an rgb() triple, and the panel ground is read from the live panel.
//    Each row also names a real element that renders in exactly that colour,
//    so the number belongs to a surface and not only to a token.
const contrast = await page.evaluate(`(() => {
  ${CONTRAST_FN}
  const panelEl = document.querySelector('.right-panel')
  const panel = parse(getComputedStyle(panelEl).backgroundColor).slice(0, 3)
  const probe = document.createElement('span')
  probe.style.display = 'none'
  panelEl.appendChild(probe)
  const resolve = (token) => {
    probe.style.color = 'var(' + token + ')'
    return parse(getComputedStyle(probe).color)
  }
  const rows = [
    ['text 1', '--text-primary', 4.5],
    ['text 2', '--text-secondary', 4.5],
    ['text 3', '--text-tertiary', 3],
    ['nominal', '--status-nominal', 3],
    ['warning', '--status-warning', 3],
    ['critical', '--status-critical', 3],
  ]
  const painted = [...panelEl.querySelectorAll('*')].filter((e) =>
    [...e.childNodes].some((n) => n.nodeType === 3 && n.textContent.trim()))
  const out = rows.map(([role, token, floor]) => {
    const fg = over(resolve(token), panel)
    const r = Math.round(ratio(fg, panel) * 100) / 100
    const want = hex(fg)
    const el = painted.find((e) => {
      const c = over(parse(getComputedStyle(e).color), panel)
      return hex(c) === want
    })
    return {
      role, token,
      fg: want, bg: hex(panel), ratio: r, floor, pass: r >= floor,
      rendered_on: el ? el.tagName.toLowerCase() + '.' + String(el.className).split(' ')[0] : '(not on this tab)',
      sample: el ? el.textContent.trim().slice(0, 26) : null,
    }
  })
  probe.remove()
  return out
})()`)
say('contrast', JSON.stringify(contrast))

// 5. A sweep of the whole document: nothing may carry a shadow, a blur, a
//    translucent surface fill, tracked caps or a radius above 2 px.
const sweep = await page.evaluate(() => {
  const all = [...document.querySelectorAll('.app *')]
  const name = (el) => `${el.tagName.toLowerCase()}.${String(el.className) || '(none)'}`
  const isSvg = (el) => el.ownerSVGElement != null || el.tagName === 'svg'
  const out = { shadow: [], blur: [], caps: [], radius: [], translucent: [] }
  for (const el of all) {
    if (isSvg(el)) continue
    const cs = getComputedStyle(el)
    if (cs.boxShadow !== 'none') out.shadow.push(name(el) + ' ' + cs.boxShadow)
    if (cs.textShadow !== 'none') out.shadow.push(name(el) + ' text ' + cs.textShadow)
    if (cs.filter.includes('blur') || cs.backdropFilter !== 'none') out.blur.push(name(el))
    if (cs.textTransform === 'uppercase') out.caps.push(name(el))
    const r = Math.max(...['borderTopLeftRadius', 'borderTopRightRadius', 'borderBottomLeftRadius', 'borderBottomRightRadius'].map((k) => parseFloat(cs[k]) || 0))
    if (r > 2 && cs.clipPath === 'none') out.radius.push(name(el) + ' ' + r + 'px')
    const bg = cs.backgroundColor.match(/[\d.]+/g)
    if (bg && bg.length > 3 && Number(bg[3]) > 0 && Number(bg[3]) < 1) out.translucent.push(name(el) + ' ' + cs.backgroundColor)
  }
  const uniq = (a) => [...new Set(a)]
  return { shadow: uniq(out.shadow), blur: uniq(out.blur), caps: uniq(out.caps), radius: uniq(out.radius), translucent: uniq(out.translucent) }
})
say('sweep', JSON.stringify(sweep))

// 6. The one headline fill, and the state marks the fold prepared for part B.
const headline = await page.evaluate(() => {
  // A state fill is a coloured, non-grey block that carries text. Chart swatches
  // and 2 px bars carry none, so they are not fills in the rule 3 sense.
  const fills = [...document.querySelectorAll('.right-panel *')].filter((el) => {
    const c = getComputedStyle(el).backgroundColor.match(/[\d.]+/g)
    if (!c) return false
    const [r, g, b, a] = [Number(c[0]), Number(c[1]), Number(c[2]), c.length > 3 ? Number(c[3]) : 1]
    if (a === 0) return false
    if (!el.textContent.trim()) return false
    return Math.max(r, g, b) - Math.min(r, g, b) > 24
  }).map((el) => `${el.tagName.toLowerCase()}.${String(el.className)} ${getComputedStyle(el).backgroundColor}`)
  // The headline fill itself: the requirement verdict in each of its states.
  const vp = document.createElement('div')
  vp.innerHTML = '<div class="requirements-verdict verdict-pass">pass</div>'
    + '<div class="requirements-verdict verdict-fail">fail</div>'
    + '<div class="requirements-verdict verdict-not-run">not run</div>'
  document.querySelector('.right-panel').appendChild(vp)
  const verdicts = [...vp.children].map((el) => ({
    cls: el.className,
    background: getComputedStyle(el).backgroundColor,
    color: getComputedStyle(el).color,
    border: getComputedStyle(el).borderTopWidth,
  }))
  vp.remove()
  const probe = document.createElement('div')
  probe.innerHTML = '<i class="state-square"></i><i class="state-square hollow"></i><i class="state-square live"></i>'
    + '<span class="prov-mark"></span><span class="prov-mark pf"></span><span class="prov-mark estimated"></span>'
    + '<span class="prov-mark probable"></span><span class="prov-mark inferred"></span><span class="prov-mark absent"></span>'
  document.querySelector('.right-panel').appendChild(probe)
  const marks = [...probe.children].map((el) => {
    const r = el.getBoundingClientRect()
    const before = getComputedStyle(el, '::before')
    return {
      cls: el.className,
      box: `${Math.round(r.width)}x${Math.round(r.height)}`,
      markWidth: before.content === 'none' ? 'none' : before.width,
      outline: before.outlineStyle === 'none' ? (getComputedStyle(el).outlineStyle === 'none' ? 'none' : getComputedStyle(el).outlineStyle) : before.outlineStyle,
      clip: before.clipPath !== 'none' ? before.clipPath : getComputedStyle(el).clipPath,
    }
  })
  probe.remove()
  return { colouredFills: fills, verdicts, marks }
})
say('headline fill and part B marks', JSON.stringify(headline))
