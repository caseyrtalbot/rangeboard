import assert from 'node:assert/strict'
import fs from 'node:fs'
import path from 'node:path'
import { createHash } from 'node:crypto'
import { chromium } from '/Users/caseytalbot/Projects/rangeboard/node_modules/playwright/index.mjs'
const [,, outDir, phase] = process.argv
const repo='/Users/caseytalbot/Projects/rangeboard'
const raw=fs.readFileSync(`${repo}/docs/field-te/evidence/onboarding-2026-09-04/firing-scenario.json`,'utf8')
fs.mkdirSync(outDir,{recursive:true})
const result={phase,date:'2026-09-07',url:'http://localhost:5443',fixtureSha256:createHash('sha256').update(raw).digest('hex'),captures:[],pageErrors:[],consoleErrors:[]}
const browser=await chromium.launch({headless:true})
result.chromium=browser.version()
const ctx=await browser.newContext({viewport:{width:1400,height:900}})
await ctx.addInitScript(()=>localStorage.setItem('rangeboard.rightpanel.width','320'))
const page=await ctx.newPage()
page.on('pageerror',e=>result.pageErrors.push(e.message))
page.on('console',m=>{if(m.type()==='error')result.consoleErrors.push(m.text())})
try {
 await page.goto(result.url)
 await page.waitForFunction(()=>window.__store && document.querySelector('.konvajs-content canvas'))
 for(const channel of ['kinetic','beam']) {
  const data=await page.evaluate(async ({raw,channel})=>{
   const {parseAutosave}=await import('/src/io/persist.ts')
   const {engagementPicture}=await import('/src/analysis/impact.ts')
   const s=parseAutosave(raw)
   if(channel==='beam') {
    const spec=s.library.find(m=>m.id==='lib-coyote').effector
    spec.effectorType='de'; spec.flyoutSpeedMps=0; spec.magazineSize=1
   }
   window.__store.getState().loadScenario(s,{fit:'always'})
   const {assessments,report}=engagementPicture(s)
   const ev=report.events[0]
   if(!ev)throw Error(`${channel}: no events`)
   const eff=s.entities.find(e=>e.id===ev.effectorId)
   const delay=s.library.find(m=>m.id===eff.specId).effector.engageTimeS
   window.__store.getState().select({entityIds:[eff.id]})
   return {ev,delay,eventCount:report.events.length}
  },{raw,channel})
  await page.click('#board-phase-analyze')
  await page.evaluate(()=>window.__store.getState().applyLayerPreset('engagement'))
  const release=data.ev.tS+data.delay
  const times=[['assigned',data.ev.tS],['preparing',release-1],['released',release],['midflight',(release+data.ev.tEffectS)/2],['effect',data.ev.tEffectS]]
  for(const [label,t] of times) {
   await page.evaluate(t=>window.__store.getState().scrubTo(t),t)
   await page.waitForTimeout(180)
   const snap=await page.evaluate(async t=>{
    const {engagementPicture}=await import('/src/analysis/impact.ts')
    const {buildTrackTimeline,shotsInFlightAt,effectorStatesAt}=await import('/src/playback/trackState.ts')
    const s=window.__store.getState().scenario
    const {assessments,report}=engagementPicture(s)
    const tl=buildTrackTimeline(s,assessments,report)
    return {t,shots:shotsInFlightAt(tl,t),effectorStates:[...effectorStatesAt(tl,t)],events:report.events}
   },t)
   const file=`${channel}-${label}.png`
   await page.screenshot({path:path.join(outDir,file),fullPage:true})
   result.captures.push({channel,label,...data,...snap,screenshot:file,rightPanelText:await page.locator('.right-panel').innerText()})
   if(phase==='after' && label==='preparing')assert.equal(snap.shots.length,0)
   if(phase==='after' && label==='released' && channel==='kinetic')assert.equal(snap.shots[0].progress,0)
  }
 }
 if(phase==='after') {
  await page.evaluate(()=>window.__store.getState().setRightTab('inspect'))
  const telemetry=page.locator('#entity-section-telemetry')
  if(await telemetry.isVisible())await telemetry.click()
  await page.locator('[data-detail="effector"]').waitFor()
  for(const id of ['magazine','release']) {
   const section=page.locator(`[data-detail-section="${id}"]`)
   const btn=section.locator('button').first()
   if(await btn.count() && await btn.getAttribute('aria-expanded')==='false')await btn.click()
  }
  await page.screenshot({path:path.join(outDir,'beam-effector-detail.png'),fullPage:true})
  await page.evaluate(()=>window.__store.getState().scrubTo(200))
  await page.waitForTimeout(150)
  result.effectorDetail=await page.locator('[data-detail="effector"]').innerText()
  await page.locator('[data-section="release"]').scrollIntoViewIfNeeded()
  await page.screenshot({path:path.join(outDir,'beam-release-detail.png'),fullPage:true})
  await page.setViewportSize({width:1280,height:800})
  await page.emulateMedia({reducedMotion:'reduce'})
  await page.locator('[data-section="release"]').scrollIntoViewIfNeeded()
  await page.screenshot({path:path.join(outDir,'beam-release-compact.png'),fullPage:true})
  result.releaseDetail=await page.locator('[data-section="release"]').innerText()
  result.panelOverflow=await page.locator('.right-panel').evaluate(el=>({scrollWidth:el.scrollWidth,clientWidth:el.clientWidth}))
  assert.ok(result.panelOverflow.scrollWidth<=result.panelOverflow.clientWidth)
  await page.setViewportSize({width:1400,height:900})
  await page.emulateMedia({reducedMotion:'no-preference'})
  result.playback=[]
  for(const speed of [1,5]) {
   await page.evaluate(async ({raw,speed})=>{
    const {parseAutosave}=await import('/src/io/persist.ts')
    window.__store.getState().loadScenario(parseAutosave(raw),{fit:'always'})
    window.__store.getState().setSpeedMult(30)
    let warm=false
    const unsub=window.__store.subscribe(st=>{
     if(warm || st.playback.simTimeS<82)return
     warm=true;unsub();window.__store.getState().pauseRun()
    })
    window.__store.getState().startRun()
   },{raw,speed})
   await page.waitForFunction(()=>window.__store.getState().playback.phase==='paused' && window.__store.getState().playback.observedThroughS>=82,null,{timeout:15000})
   await page.evaluate(speed=>{
    window.__store.getState().scrubTo(78)
    window.__store.getState().setSpeedMult(speed)
   },speed)
   await page.click('#board-phase-execute')
   await page.evaluate(()=>{
    window.__r3samples=[]
    let done=false
    const unsub=window.__store.subscribe(st=>{
     if(done)return
     window.__r3samples.push({t:st.playback.simTimeS,playing:st.playback.playing})
     if(st.playback.simTimeS>=82){done=true;unsub();window.__store.getState().pauseRun()}
    })
    window.__store.getState().startRun()
   })
   await page.waitForFunction(()=>window.__store.getState().playback.phase==='paused' && window.__store.getState().playback.simTimeS>=82,null,{timeout:10000})
   result.playback.push({speed,samples:await page.evaluate(()=>window.__r3samples)})
   await page.screenshot({path:path.join(outDir,`execute-${speed}x.png`),fullPage:true})
  }
 }
 await page.setViewportSize({width:1280,height:800})
 await page.emulateMedia({reducedMotion:'reduce'})
 await page.screenshot({path:path.join(outDir,'compact-reduced-motion.png'),fullPage:true})
 assert.deepEqual(result.pageErrors,[])
 assert.deepEqual(result.consoleErrors,[])
 result.status='ok'
} catch(e) {result.status='failed';result.error=String(e);process.exitCode=1}
finally {fs.writeFileSync(path.join(outDir,'r3-capture.json'),JSON.stringify(result,null,2)+'\n');await ctx.close();await browser.close()}
console.log(JSON.stringify({status:result.status,error:result.error,captures:result.captures.length,pageErrors:result.pageErrors}))
