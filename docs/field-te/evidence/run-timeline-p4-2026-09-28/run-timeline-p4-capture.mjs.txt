/** Play for `ms` and record every requestAnimationFrame interval and long task. */
const playAndRecord = (ms) =>
  page.evaluate(
    (ms) =>
      new Promise((resolve) => {
        const longtasks = []
        let obs = null
        try {
          obs = new PerformanceObserver((list) => {
            for (const e of list.getEntries()) longtasks.push({ startMs: e.startTime, durationMs: e.duration })
          })
          obs.observe({ type: 'longtask', buffered: false })
        } catch {
          obs = null
        }
        const st = window.__store.getState()
        st.setPlaying(true)
        const gaps = []
        let last = null
        const stop = performance.now() + ms
        function frame(t) {
          if (last != null) gaps.push(t - last)
          last = t
          if (t < stop) requestAnimationFrame(frame)
          else {
            window.__store.getState().setPlaying(false)
            setTimeout(() => {
              obs?.disconnect()
              resolve({ gaps, longtasks, longtaskSupported: obs != null })
            }, 50)
          }
        }
        requestAnimationFrame(frame)
      }),
    ms,
  )

// over16ms counts intervals over 16.7 ms, one 60 Hz frame.
function frameStats({ gaps, longtasks, longtaskSupported }) {
  const sorted = [...gaps].sort((a, b) => a - b)
  return {
    intervals: gaps.length,
    over16ms: gaps.filter((g) => g > 16.7).length,
    over20ms: gaps.filter((g) => g > 20).length,
    over33ms: gaps.filter((g) => g > 33).length,
    maxMs: r2(sorted[sorted.length - 1]),
    meanMs: r2(gaps.reduce((a, b) => a + b, 0) / gaps.length),
    p99Ms: r2(sorted[Math.floor(sorted.length * 0.99)]),
    longtaskSupported,
    longtasks: longtasks.length,
    longtaskMaxMs: longtasks.length ? r2(Math.max(...longtasks.map((l) => l.durationMs))) : 0,
    longtaskTotalMs: r2(longtasks.reduce((a, l) => a + l.durationMs, 0)),
  }
}

// 1. Frames: open and expanded from T+0, closed from T+0, then open again from T+0.
const frames = { speed: 30, recordMs: 10000, runs: [] }
async function frameRun(label, open) {
  await scrub(0)
  const isOpen = await page.evaluate(() => !!document.querySelector('#run-timeline'))
  if (isOpen !== open) {
    await page.click('#run-timeline-toggle')
    await sleep(400)
  }
  if (open) await expandAll()
  const before = await clock()
  const domBefore = open ? await dom() : null
  const rec = await playAndRecord(10000)
  const after = await clock()
  const domAfter = open ? await dom() : null
  const run = {
    label,
    strip: open ? 'open, every group and sensor row expanded' : 'closed',
    ...frameStats(rec),
    simTimeS: [r4(before.simTimeS), r4(after.simTimeS)],
    observedThroughS: [r4(before.observedThroughS), r4(after.observedThroughS)],
    domBefore,
    domAfter,
  }
  frames.runs.push(run)
  console.log('frames', JSON.stringify(run))
  return run
}
await frameRun('open-1', true)
await frameRun('closed-1', false)
await frameRun('open-2', true)
await frameRun('closed-2', false)
const opens = frames.runs.filter((r) => r.label.startsWith('open'))
const closeds = frames.runs.filter((r) => r.label.startsWith('closed'))
frames.pass = opens.every((o) => o.over33ms <= Math.max(...closeds.map((c) => c.over33ms)))
frames.rule = 'Phase 2 rule: an open run adds no interval over 33 ms beyond the closed runs'
findings.frames = frames

// 4. runTimelineOf per horizon on a fresh view copy: the first build, each pass,
// and how many row objects are new or reused against the previous pass.
findings.modelPasses = await page.evaluate(async () => {
  const { selectTrialSelection } = await import('/src/state/trialSelectors.ts')
  const { trialResultOf } = await import('/src/analysis/trialResult.ts')
  const { runTimelineOf } = await import('/src/analysis/runTimeline.ts')
  const s = window.__store.getState()
  const sel = selectTrialSelection(s)
  const base = trialResultOf(sel.scenario, sel.result.manifest, sel.result)
  const view = { ...base }
  const allRows = (m) => m.groups.flatMap((g) => g.rows.flatMap((r) => [r, ...(r.subLanes ?? [])]))
  const round = (x) => Math.round(x * 1000) / 1000
  const t0 = performance.now()
  const first = runTimelineOf(view, 0)
  const firstBuildMs = round(performance.now() - t0)
  const passes = []
  let prev = new Map(allRows(first).map((r) => [r.key, r]))
  const horizons = []
  for (let h = 5; h <= view.durationS; h += 5) horizons.push(h)
  if (horizons[horizons.length - 1] !== view.durationS) horizons.push(view.durationS)
  for (const h of horizons) {
    const a = performance.now()
    const m = runTimelineOf(view, h)
    const ms = performance.now() - a
    const rows = allRows(m)
    let reused = 0
    for (const r of rows) if (prev.get(r.key) === r) reused++
    passes.push({ horizonS: h, ms: round(ms), rows: rows.length, reused, created: rows.length - reused })
    prev = new Map(rows.map((r) => [r.key, r]))
  }
  const ms = passes.map((p) => p.ms).sort((a, b) => a - b)
  const sum = (k) => passes.reduce((a, p) => a + p[k], 0)
  return {
    durationS: view.durationS,
    observedThroughS: view.observedThroughS,
    firstBuildMs,
    passes: passes.length,
    passMeanMs: round(sum('ms') / passes.length),
    passMedianMs: ms[Math.floor(ms.length / 2)],
    passMaxMs: ms[ms.length - 1],
    rowsPerPassMax: Math.max(...passes.map((p) => p.rows)),
    createdTotal: sum('created'),
    reusedTotal: sum('reused'),
    createdMeanPerPass: round(sum('created') / passes.length),
    perPass: passes,
  }
})
console.log('model passes', JSON.stringify({ ...findings.modelPasses, perPass: undefined }))

const contrast = await (async () => {
  const out = { rule: INKS.rule, themes: {} }
  for (const theme of ['dark', 'light']) {
    await page.evaluate((t) => window.__store.getState().setTheme(t), theme)
    await sleep(400)
    out.themes[theme] = await page.evaluate((inks) => {
      const root = document.querySelector('#run-timeline')
      const probe = document.createElement('span')
      probe.style.display = 'none'
      root.appendChild(probe)
      const parse = (c) => {
        let m = c.match(/^rgba?\(([\d.]+),\s*([\d.]+),\s*([\d.]+)(?:,\s*([\d.]+))?\)$/)
        if (m) return { r: +m[1], g: +m[2], b: +m[3], a: m[4] == null ? 1 : +m[4] }
        m = c.match(/^color\(srgb ([\d.e-]+) ([\d.e-]+) ([\d.e-]+)(?: \/ ([\d.]+))?\)$/)
        if (m) return { r: +m[1] * 255, g: +m[2] * 255, b: +m[3] * 255, a: m[4] == null ? 1 : +m[4] }
        throw new Error(`unparsed colour ${c}`)
      }
      const resolve = (expr) => {
        probe.style.color = ''
        probe.style.color = expr
        if (!probe.style.color) throw new Error(`invalid expression ${expr}`)
        return parse(getComputedStyle(probe).color)
      }
      const over = (fg, bg, alpha) => {
        const a = alpha * fg.a
        return { r: fg.r * a + bg.r * (1 - a), g: fg.g * a + bg.g * (1 - a), b: fg.b * a + bg.b * (1 - a), a: 1 }
      }
      const lum = ({ r, g, b }) => {
        const f = (v) => {
          v /= 255
          return v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4
        }
        return 0.2126 * f(r) + 0.7152 * f(g) + 0.0722 * f(b)
      }
      const ratio = (a, b) => {
        const [x, y] = [lum(a), lum(b)].sort((p, q) => q - p)
        return (x + 0.05) / (y + 0.05)
      }
      const hex = ({ r, g, b }) =>
        '#' + [r, g, b].map((v) => Math.round(v).toString(16).padStart(2, '0')).join('')
      const surfaces = Object.fromEntries(Object.entries(inks.surfaces).map(([k, e]) => [k, resolve(e)]))
      const rows = []
      for (const ink of inks.items) {
        let fg = null
        let source = 'expression'
        if (ink.sel) {
          const el = root.querySelector(ink.sel)
          if (el) {
            const cs = getComputedStyle(el)
            const v = cs[ink.prop]
            if (v && v !== 'none') {
              fg = parse(v)
              source = `element ${ink.sel} ${ink.prop}`
            }
          }
        }
        if (!fg) fg = resolve(ink.expr)
        for (const on of ink.on) {
          const bg = surfaces[on]
          const shown = over(fg, bg, ink.alpha ?? 1)
          const value = Math.round(ratio(shown, bg) * 100) / 100
          rows.push({
            id: ink.id,
            role: ink.role,
            ink: hex(fg),
            alpha: ink.alpha ?? 1,
            shown: hex(shown),
            surface: on,
            surfaceColour: hex(bg),
            source,
            ratio: value,
            required: ink.required,
            pass: ink.required == null ? null : value >= ink.required,
          })
        }
      }
      probe.remove()
      return { surfaces: Object.fromEntries(Object.entries(surfaces).map(([k, v]) => [k, hex(v)])), rows }
    }, INKS)
  }
  return out
})()
for (const [t, v] of Object.entries(contrast.themes)) {
  const fails = v.rows.filter((r) => r.pass === false)
  contrast[`${t}Summary`] = {
    pairs: v.rows.length,
    required: v.rows.filter((r) => r.pass != null).length,
    fail: fails.length,
    failing: fails.map((r) => `${r.id} on ${r.surface} ${r.ratio}`),
  }
  console.log('contrast', t, JSON.stringify(contrast[`${t}Summary`]))
}
