/** Store facts and the viewport on screen now (`shownViewport`, as the handlers read it). */
const st = () =>
  page.evaluate(async () => {
    const { clampViewport, DEFAULT_WINDOW_S } = await import('/src/charts/timeViewport.ts')
    const { shownViewport } = await import('/src/panels/timeline/useShownT0.ts')
    const { durationOf } = await import('/src/state/selectors.ts')
    const { useStore } = await import('/src/state/store.ts')
    const s = window.__store.getState()
    const rt = s.ui.runTimeline
    const durationS = durationOf(s.scenario)
    const base = clampViewport(rt.viewport ?? { t0S: 0, winS: DEFAULT_WINDOW_S }, durationS)
    return {
      sameStore: useStore === window.__store,
      simTimeS: s.playback.simTimeS,
      observedThroughS: s.playback.observedThroughS,
      playing: s.playback.playing,
      follow: rt.follow,
      base,
      shown: shownViewport(base, rt.follow, durationS),
      durationS,
      openGroups: rt.openGroups,
      openSensors: rt.openSensors,
      entityIds: s.selection.entityIds,
    }
  })

/** The run-time model the strip draws, and the plot scale. */
const modelFacts = () =>
  page.evaluate(async () => {
    const { selectTrialHorizon, selectTrialSelection } = await import('/src/state/trialSelectors.ts')
    const { trialResultOf } = await import('/src/analysis/trialResult.ts')
    const { runTimelineOf } = await import('/src/analysis/runTimeline.ts')
    const s = window.__store.getState()
    const sel = selectTrialSelection(s)
    const view = trialResultOf(sel.scenario, sel.result.manifest, sel.result)
    const horizonS = selectTrialHorizon(s, s.ui.boardPhase === 'analyze' ? Infinity : s.playback.simTimeS)
    const model = runTimelineOf(view, horizonS)
    const rows = model.groups.flatMap((g) => g.rows)
    return {
      groups: model.groups.map((g) => g.key),
      subLaneCount: rows.reduce((n, r) => n + (r.subLanes?.length ?? 0), 0),
      rows: rows.map((r) => ({ key: r.key, kind: r.kind, label: r.label, entityId: r.entityId })),
      markTimes: rows.flatMap((r) => [...r.marks, ...(r.subLanes ?? []).flatMap((l) => l.marks)].map((m) => m.tS)),
    }
  })

/** On-screen mark glyphs inside the visible plot, with their run times (cx / pxPerS, as Phase 2). */
const visibleMarks = () =>
  page.evaluate(() => {
    const plot = document.querySelector('.runtl-plotcol').getBoundingClientRect()
    const scroll = document.querySelector('.runtl-scroll').getBoundingClientRect()
    const pan = document.querySelector('.runtl-pan').getBoundingClientRect()
    const svg = document.querySelector('.runtl-svg')
    const pxPerS = Number(document.querySelector('.runtl-plotcol').clientWidth) / window.__p3winS
    const OFFSET = { release: 3.5, drop: 3 }
    return [...svg.querySelectorAll('.runtl-m')]
      .map((el) => {
        const kind = [...el.classList].find((c) => c.startsWith('runtl-m-')).slice(8)
        let cx
        if (el.tagName === 'circle') cx = Number(el.getAttribute('cx'))
        else if (el.tagName === 'line') cx = Number(el.getAttribute('x1'))
        else cx = Number(el.getAttribute('d').slice(1).split(' ')[0]) + (OFFSET[kind] ?? 0)
        const b = el.getBoundingClientRect()
        return { kind, tS: cx / pxPerS, x: pan.left + cx, y: b.top + b.height / 2 }
      })
      .filter((m) => m.x > plot.left + 20 && m.x < plot.right - 20 && m.y > scroll.top + 4 && m.y < scroll.bottom - 4)
  })

const setWin = async () => {
  const s = await st()
  await page.evaluate((w) => (window.__p3winS = w), s.base.winS)
}

async function shot(name, state) {
  await sleep(300)
  const file = join(OUT, `${name}.png`)
  await page.screenshot({ path: file })
  const s = await st()
  findings.shots.push({ file: `${name}.png`, state, simTimeS: s.simTimeS, observedThroughS: s.observedThroughS })
  console.log('shot', name)
}

/** Runs one interaction and records observedThroughS before and after. */
async function item(n, name, fn) {
  const before = await st()
  const result = await fn(before)
  const after = await st()
  findings.items[n] = {
    name,
    observedBeforeS: before.observedThroughS,
    observedAfterS: after.observedThroughS,
    observedUnchanged: before.observedThroughS === after.observedThroughS,
    ...result,
  }
  console.log(n, name, JSON.stringify(findings.items[n]))
}

}

// 4. Tooltip: hover a mark, read it inside #run-timeline, Escape closes it.
await item(4, 'tooltip and Escape', async () => {
  const marks = await visibleMarks()
  const m = marks[Math.floor(marks.length / 2)]
  await page.mouse.move(m.x, m.y)
  await sleep(200)
  const inside = await page.evaluate(() => !!document.querySelector('#run-timeline .runtl-tip'))
  const text = await tipText()

  await page.keyboard.press('Escape')
  await sleep(150)

// 4b. Escape closes the tip and keeps the board selection; a 1 px move over the same
// mark does not reopen it; a move to another mark does.
await item('4b', 'Escape keeps the selection and the closed tip', async () => {
  // Leave the mark item 4 closed, so the next hover over it opens the tip again.
  await page.mouse.move(5, 5)
  await sleep(150)
  const facts = await modelFacts()
  const entity = facts.rows.find((r) => r.entityId != null)
  await page.evaluate((id) => window.__store.getState().select({ entityIds: [id] }), entity.entityId)
  await sleep(300)
  const selectedBefore = (await st()).entityIds
  const marks = await visibleMarks()
  const m = marks[Math.floor(marks.length / 2)]
  const other = marks.find((k) => Math.hypot(k.x - m.x, k.y - m.y) > 20)
  await page.mouse.move(m.x - 1, m.y)
  await page.mouse.move(m.x, m.y)
  await sleep(200)
  const tipOpen = (await tipText()) !== null
  await page.keyboard.press('Escape')
  await sleep(150)
  const tipAfterEscape = (await tipText()) !== null
  const selectedAfter = (await st()).entityIds
  await page.mouse.move(m.x + 1, m.y)
  await sleep(200)
  const tipAfterSmallMove = (await tipText()) !== null
  await page.mouse.move(other.x, other.y)
  await sleep(200)
  const tipOnOther = (await tipText()) !== null
  await page.mouse.move(5, 5)
  await page.evaluate(() => window.__store.getState().select({ entityIds: [] }))
  await sleep(150)
  return {
    entity: entity.label,
    mark: { kind: m.kind, tS: r4(m.tS) },
    otherMark: { kind: other.kind, tS: r4(other.tS) },
    tipOpen,
    tipAfterEscape,
    selectedBefore,
    selectedAfter,
    selectionKept: JSON.stringify(selectedBefore) === JSON.stringify(selectedAfter) && selectedAfter.length === 1,
    tipAfterSmallMove,
    tipOnOther,
  }
})

const selectedLines = () =>
  page.evaluate(() => {
    const sr = document.querySelector('.runtl-scroll').getBoundingClientRect()
    return [...document.querySelectorAll('.runtl-gutter .runtl-line.selected')].map((l) => {
      const r = l.getBoundingClientRect()

    })
  })
/** The gutter row line (`t: 'row'`) of a label, its selection and whether it is in view. */
const rowLine = (label) =>
  page.evaluate((label) => {
    const scroll = document.querySelector('.runtl-scroll')
    const sr = scroll.getBoundingClientRect()
    const line = [...document.querySelectorAll('.runtl-gutter .runtl-row')].find((l) =>
      l.querySelector(`.runtl-select[title="${label}"]`),
    )
    const lr = line?.getBoundingClientRect()
    return {
      scrollTop: scroll.scrollTop,
      selectedLines: document.querySelectorAll('.runtl-gutter .runtl-line.selected').length,
      selectedGrounds: document.querySelectorAll('.runtl-ground-selected').length,
      rowSelected: !!line && line.classList.contains('selected'),
      rowInView: !!lr && lr.top >= sr.top - 0.5 && lr.bottom <= sr.bottom + 0.5,
    }
  }, label)

  const b = await st()
  await page.evaluate(() => window.__store.getState().playbackCommand({ type: 'play' }))
  await sleep(100)
  const a = await st()
  await page.evaluate(() => window.__store.getState().pauseRun())
  await sleep(200)
  return { followBefore: b.follow, followAfterPlay: a.follow, playing: a.playing, note: 'Play advances observation by design.' }
})

// 15. Reduced motion: no 100 ms glide on the four gliding elements.
await item(15, 'reduced motion', async () => {
  const SEL = ['.runtl-pan', '.runtl-playhead', '.runtl-axis-svg > g', '.runtl-over-window']
  const read = () =>
    page.evaluate((sel) => Object.fromEntries(sel.map((s) => [s, getComputedStyle(document.querySelector(s)).transition])), SEL)
  const follow = (await st()).follow
  const normal = await read()
  await page.emulateMedia({ reducedMotion: 'reduce' })
  await sleep(200)
  const reduced = await read()
  await page.emulateMedia({ reducedMotion: 'no-preference' })
  return {
    follow,
    normal,
    reduced,
    glideWithoutReduce: Object.values(normal).some((t) => t.includes('0.1s')),
    noGlideWithReduce: Object.values(reduced).every((t) => !t.includes('0.1s')),
  }
})

// 17. The hide control hides a row; Show all shows it again.
await item(17, 'hide a row and Show all', async () => {
  const hidden = async () => page.evaluate(() => window.__store.getState().ui.runTimeline.hiddenRows)
  const showAll = () => page.locator('.runtl-head button.mini', { hasText: 'Show all' })
  const row = page.locator('.runtl-gutter .runtl-row').filter({ has: page.locator('.runtl-hide') }).first()
  const label = await row.locator('.runtl-select, .runtl-name').first().getAttribute('title')
  const facts = await modelFacts()
  const key = facts.rows.find((r) => r.label === label)?.key
  const hiddenBefore = await hidden()
  const showAllBefore = await showAll().count()
  await row.hover()
  await row.locator('.runtl-hide').click()
  await sleep(200)
  const hiddenAfterHide = await hidden()
  const showAllText = (await showAll().count()) > 0 ? await showAll().textContent() : null
  const rowDrawn = await page.evaluate(
    (label) => [...document.querySelectorAll('.runtl-gutter .runtl-row [title]')].some((e) => e.title === label),
    label,
  )
  await showAll().click()
  await sleep(200)
  const hiddenAfterShowAll = await hidden()
  const showAllAfter = await showAll().count()
  await page.mouse.move(5, 5)
  return {
    label,
    rowKey: key,
    hiddenBefore,
    showAllBefore,
    hiddenAfterHide,
    hidGainsRow: hiddenAfterHide.includes(key),
    rowDrawnAfterHide: rowDrawn,
    showAllText,
    hiddenAfterShowAll,
    showAllAfter,
    cleared: hiddenAfterShowAll.length === 0 && showAllAfter === 0,
  }
})
