// F1 render-count and horizon-parity probe (the baseline probe of
// f1-cadence-2026-09-05 plus the horizon checks of brief section 6.2). Counts
// committed component renders per React renderer (react-dom, react-konva)
// through the DevTools global hook while the verified firing fixture plays at
// fixed speeds, records every publication's observed horizon, and runs the
// last seconds of the run to completion at each speed.
// Usage: node f1-render-probe.mjs <baseUrl> <outDir> [headless|headed]
import { createRequire } from 'node:module'
const { chromium } = createRequire('/Users/caseytalbot/Projects/rangeboard/package.json')('playwright')
import { writeFileSync, mkdirSync } from 'node:fs'
import { join } from 'node:path'

const [baseUrl = 'http://localhost:5301/', outDir = '.', mode = 'headless'] = process.argv.slice(2)
mkdirSync(outDir, { recursive: true })

const SPEEDS = [1, 5, 10, 20, 30]
const WINDOW_MS = 6000
const FIXTURE = '/docs/field-te/evidence/onboarding-2026-09-04/firing-scenario.json'

// Installed before any script runs: a minimal DevTools hook that counts, per commit,
// every function/class/memo/forwardRef fiber carrying the PerformedWork flag.
const HOOK = `(() => {
  const PerformedWork = 1
  const COMPONENT_TAGS = new Set([0, 1, 11, 14, 15])
  const renderers = new Map()
  const state = {
    recording: false,
    frames: 0,
    perRenderer: {},
  }
  function fresh() {
    return { commits: 0, renderMs: 0, renders: {}, inclusiveMs: {}, firstCommits: [] }
  }
  function nameOf(f) {
    const t = f.type
    if (typeof t === 'function') return t.displayName || t.name || 'anonymous'
    if (t && typeof t === 'object') {
      if (typeof t.render === 'function') return t.render.displayName || t.render.name || 'forwardRef'
      if (t.type) return nameOf({ type: t.type })
    }
    return String(t)
  }
  function walk(fiber, bucket, ms, sample) {
    const alt = fiber.alternate
    if (COMPONENT_TAGS.has(fiber.tag) && (fiber.flags & PerformedWork)) {
      const n = nameOf(fiber)
      bucket[n] = (bucket[n] || 0) + 1
      if (typeof fiber.actualDuration === 'number') ms[n] = (ms[n] || 0) + fiber.actualDuration
      if (sample) sample[n] = (sample[n] || 0) + 1
    }
    if (alt == null || fiber.child !== alt.child) {
      for (let c = fiber.child; c; c = c.sibling) walk(c, bucket, ms, sample)
    }
  }
  let nextId = 0
  const hook = {
    supportsFiber: true,
    isDisabled: false,
    renderers,
    inject(internals) {
      const id = ++nextId
      renderers.set(id, internals)
      return id
    },
    on() {}, off() {}, emit() {},
    onScheduleFiberRoot() {},
    onCommitFiberUnmount() {},
    onPostCommitFiberRoot() {},
    onCommitFiberRoot(id, root) {
      if (!state.recording) return
      const key = id === window.__f1.konvaId ? 'react-konva'
        : (renderers.get(id) || {}).rendererPackageName || ('renderer-' + id)
      const s = (state.perRenderer[key] ||= fresh())
      s.commits += 1
      const cur = root.current
      if (typeof cur.actualDuration === 'number') s.renderMs += cur.actualDuration
      const sample = s.firstCommits.length < 12 ? {} : null
      walk(cur, s.renders, s.inclusiveMs, sample)
      if (sample) s.firstCommits.push({ t: Math.round(performance.now() - state.t0), ms: +cur.actualDuration.toFixed(2), sample })
    },
  }
  Object.defineProperty(window, '__REACT_DEVTOOLS_GLOBAL_HOOK__', { value: hook, configurable: true })
  const raf = window.requestAnimationFrame.bind(window)
  window.requestAnimationFrame = (cb) => raf((t) => { if (state.recording) state.frames += 1; cb(t) })
  window.__f1 = {
    state,
    hook,
    start() { state.recording = true; state.frames = 0; state.perRenderer = {}; state.t0 = performance.now() },
    stop() { state.recording = false; return JSON.parse(JSON.stringify({ frames: state.frames, perRenderer: state.perRenderer })) },
  }
  try { localStorage.setItem('rangeboard.rightpanel.width', '320') } catch {}
})()`

const browser = await chromium.launch({ headless: mode !== 'headed' })
const context = await browser.newContext({
  viewport: { width: 1400, height: 900 },
  deviceScaleFactor: 2,
})
await context.addInitScript(HOOK)
const page = await context.newPage()
const consoleLines = []
const pageErrors = []
page.on('console', (m) => consoleLines.push(`[${m.type()}] ${m.text()}`))
page.on('pageerror', (e) => pageErrors.push(e.message))
page.on('dialog', (d) => d.accept())

await page.goto(baseUrl, { waitUntil: 'networkidle' })
await page.waitForFunction(() => Boolean(window.__store))

// Hook the Konva reconciler through the same DevTools path. react-konva does not
// inject itself, so the probe imports the app's own module instance and injects it.
const setup = await page.evaluate(async (fixture) => {
  const { parseAutosave } = await import('/src/io/persist.ts')
  const { scenarioDurationS } = await import('/src/model/scenario.ts')
  const konvaUrl = performance
    .getEntriesByType('resource')
    .map((e) => e.name)
    .find((n) => /react-konva[^/]*\.js/.test(n))
  let konvaInjected = false
  if (konvaUrl) {
    const mod = await import(konvaUrl)
    const r = mod.KonvaRenderer
    if (r && typeof r.injectIntoDevTools === 'function') {
      konvaInjected = Boolean(
        r.injectIntoDevTools({ bundleType: 1, version: mod.version, rendererPackageName: 'react-konva' }),
      )
      window.__f1.konvaId = window.__f1.hook.renderers.size
    }
  }
  const raw = await (await fetch(fixture)).text()
  const s = window.__store.getState()
  s.loadScenario(parseAutosave(raw))
  s.patchScenario({ runMode: 'deterministic' })
  s.applyLens('run-summary')
  s.clearSelection()
  window.__f1.pubs = { simTime: 0, any: 0, times: [], horizon: [] }
  window.__store.subscribe((n, p) => {
    if (!window.__f1.state.recording) return
    window.__f1.pubs.any += 1
    if (n.playback.simTimeS !== p.playback.simTimeS) {
      window.__f1.pubs.simTime += 1
      window.__f1.pubs.times.push(performance.now())
    }
    if (n.playback !== p.playback) {
      // Horizon parity: while a run is live, every publication's observed
      // horizon equals the published time and never moves backward.
      window.__f1.pubs.horizon.push({
        t: n.playback.simTimeS,
        obs: n.playback.observedThroughS,
        prevObs: p.playback.observedThroughS,
        phase: n.playback.phase,
      })
    }
  })
  const st = window.__store.getState()
  return {
    konvaUrl,
    konvaInjected,
    renderers: [...window.__f1.hook.renderers.values()].map((r) => `${r.rendererPackageName}@${r.version}`),
    durationS: scenarioDurationS(st.scenario),
    entities: st.scenario.entities.map((e) => ({ id: e.id, kind: e.kind, name: e.name })),
    runMode: st.scenario.runMode,
    dpr: window.devicePixelRatio,
    ua: navigator.userAgent,
  }
}, FIXTURE)

const uasId = setup.entities.find((e) => e.kind === 'uas')?.id

async function measure(label, { tab, select, speed, runMode, windowMs = WINDOW_MS }) {
  await page.evaluate(
    ({ tab, select, speed, runMode }) => {
      const s = window.__store.getState()
      s.resetRun('manual')
      s.setRightTab(tab)
      if (select) s.select({ entityIds: [select] })
      else s.clearSelection()
      s.setSpeedMult(speed)
      if (runMode && s.scenario.runMode !== runMode) s.patchScenario({ runMode })
    },
    { tab, select, speed, runMode },
  )
  await page.waitForTimeout(500)
  const result = await page.evaluate(async (windowMs) => {
    const sleep = (ms) => new Promise((r) => setTimeout(r, ms))
    const pubs = window.__f1.pubs
    pubs.simTime = 0
    pubs.any = 0
    pubs.times = []
    pubs.horizon = []
    window.__f1.start()
    const t0 = performance.now()
    const st = window.__store.getState()
    if (windowMs > 0) st.startRun()
    let phase = window.__store.getState().playback.phase
    while (performance.now() - t0 < windowMs && phase !== 'complete') {
      await sleep(50)
      phase = window.__store.getState().playback.phase
    }
    if (windowMs > 0 && phase !== 'complete') window.__store.getState().pauseRun()
    if (windowMs === 0) await sleep(3000)
    const wallMs = performance.now() - t0
    const counts = window.__f1.stop()
    const p = window.__store.getState().playback
    const gaps = pubs.times.slice(1).map((t, i) => t - pubs.times[i]).sort((a, b) => a - b)
    const q = (f) => (gaps.length ? +gaps[Math.min(gaps.length - 1, Math.floor(f * gaps.length))].toFixed(2) : null)
    const live = pubs.horizon.filter((h) => h.phase === 'running' || h.phase === 'complete')
    const horizon = {
      samples: live.length,
      observedNotEqualPublished: live.filter((h) => h.obs !== h.t).length,
      observedDecreased: pubs.horizon.filter((h) => h.obs < h.prevObs).length,
      finalSimTimeS: p.simTimeS,
      finalObservedThroughS: p.observedThroughS,
    }
    return {
      wallMs,
      simElapsedS: p.simTimeS,
      phase: p.phase,
      speedMult: p.speedMult,
      publications: { simTime: pubs.simTime, any: pubs.any, gapMs: { min: q(0), p50: q(0.5), p90: q(0.9), max: q(0.999) } },
      horizon,
      ...counts,
    }
  }, windowMs)
  return { label, tab, select: select ?? null, speed, runMode: runMode ?? setup.runMode, ...result }
}

const cells = []
cells.push(await measure('idle inspect', { tab: 'inspect', select: uasId, speed: 1, windowMs: 0 }))
cells.push(await measure('idle analysis', { tab: 'analysis', speed: 1, windowMs: 0 }))
for (const tab of ['inspect', 'analysis']) {
  for (const speed of SPEEDS) {
    cells.push(
      await measure(`${tab} ${speed}x`, { tab, select: tab === 'inspect' ? uasId : undefined, speed }),
    )
  }
}
cells.push(await measure('analysis 1x monte-carlo', { tab: 'analysis', speed: 1, runMode: 'monte-carlo' }))

// Completion parity (brief 6.2): the last seconds of the run at each speed, so
// the loop's own advanceRun/completeRun path seals the run at durationS exactly.
const completion = []
for (const speed of [1, 5, 10, 30]) {
  const r = await page.evaluate(async ({ speed, durationS }) => {
    const sleep = (ms) => new Promise((r) => setTimeout(r, ms))
    const s = window.__store.getState()
    if (s.scenario.runMode !== 'deterministic') s.patchScenario({ runMode: 'deterministic' })
    s.resetRun('manual')
    s.setRightTab('inspect')
    s.clearSelection()
    s.setSpeedMult(speed)
    s.startRun()
    s.advanceRun(durationS - 3)
    s.pauseRun()
    const pubs = window.__f1.pubs
    pubs.horizon = []
    window.__f1.start()
    const t0 = performance.now()
    window.__store.getState().startRun()
    while (performance.now() - t0 < 8000 && window.__store.getState().playback.phase !== 'complete') await sleep(25)
    window.__f1.stop()
    const p = window.__store.getState().playback
    const live = pubs.horizon.filter((h) => h.phase === 'running' || h.phase === 'complete')
    return {
      speed,
      wallMs: +(performance.now() - t0).toFixed(0),
      phase: p.phase,
      playing: p.playing,
      simTimeS: p.simTimeS,
      observedThroughS: p.observedThroughS,
      durationS,
      completedAtDuration: p.phase === 'complete' && p.simTimeS === durationS && p.observedThroughS === durationS,
      publications: live.length,
      observedNotEqualPublished: live.filter((h) => h.obs !== h.t).length,
      observedDecreased: pubs.horizon.filter((h) => h.obs < h.prevObs).length,
      maxPublishedS: Math.max(...live.map((h) => h.t)),
    }
  }, { speed, durationS: setup.durationS })
  completion.push(r)
}
await page.evaluate(() => {
  const s = window.__store.getState()
  s.patchScenario({ runMode: 'deterministic' })
  s.resetRun('manual')
  s.setRightTab('inspect')
  s.setSpeedMult(1)
  s.startRun()
})
await page.waitForTimeout(2500)
await page.evaluate(() => window.__store.getState().pauseRun())
await page.screenshot({ path: join(outDir, 'f1-after-inspect.png') })

const out = {
  date: new Date().toISOString(),
  mode,
  baseUrl,
  fixture: FIXTURE,
  viewport: { width: 1400, height: 900, deviceScaleFactor: 2, rightPanelPx: 320 },
  windowMs: WINDOW_MS,
  setup,
  cells,
  completion,
  pageErrors,
}
writeFileSync(join(outDir, 'render-counts.json'), JSON.stringify(out, null, 2))
writeFileSync(join(outDir, 'browser-console.log'), consoleLines.join('\n') + '\n')
await browser.close()

// Short table on stdout.
for (const c of cells) {
  const dom = c.perRenderer['react-dom'] || { commits: 0, renderMs: 0, renders: {} }
  const konva = c.perRenderer['react-konva'] || { commits: 0, renderMs: 0, renders: {} }
  const pick = (r, n) => r.renders[n] || 0
  console.log(
    [
      c.label.padEnd(24),
      `wall ${(c.wallMs / 1000).toFixed(2)}s`,
      `sim ${c.simElapsedS.toFixed(1)}s`,
      `frames ${c.frames}`,
      `pubs ${c.publications.simTime}/${c.publications.any} gap p50 ${c.publications.gapMs.p50}ms`,
      `horizon neq ${c.horizon.observedNotEqualPublished} dec ${c.horizon.observedDecreased}`,
      `dom commits ${dom.commits} (${dom.renderMs.toFixed(0)} ms)`,
      `konva commits ${konva.commits} (${konva.renderMs.toFixed(0)} ms)`,
      `TimelineBar ${pick(dom, 'TimelineBar')}`,
      `CanvasStage ${pick(dom, 'CanvasStage')}`,
      `Inspector ${pick(dom, 'Inspector')}`,
      `AnalysisPanel ${pick(dom, 'AnalysisPanel')}`,
      `EntityLayer ${pick(konva, 'EntityLayer')}`,
    ].join(' | '),
  )
}
for (const r of completion) {
  console.log(
    `completion ${r.speed}x | phase ${r.phase} | sim ${r.simTimeS} obs ${r.observedThroughS} of ${r.durationS} | exact ${r.completedAtDuration} | pubs ${r.publications} neq ${r.observedNotEqualPublished} dec ${r.observedDecreased} | wall ${r.wallMs}ms`,
  )
}
if (pageErrors.length) {
  console.error('page errors', pageErrors)
  process.exitCode = 1
}
