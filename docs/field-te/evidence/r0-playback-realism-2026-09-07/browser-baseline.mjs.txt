import assert from 'node:assert/strict'
import fs from 'node:fs'
import path from 'node:path'
import { createHash } from 'node:crypto'
import { execFileSync } from 'node:child_process'
import { chromium } from '/Users/caseytalbot/Projects/rangeboard/node_modules/playwright/index.mjs'

const repo = '/Users/caseytalbot/Projects/rangeboard'
const [,, fixture = '/private/tmp/r0-perf/scenario.json', url = 'http://localhost:5391',
  outDir = `${repo}/docs/field-te/evidence/r0-playback-realism-2026-09-07`] = process.argv
assert.equal(new URL(url).origin, 'http://localhost:5391', 'Use only the owned dev origin')
const raw = fs.readFileSync(fixture, 'utf8')
const lock = JSON.parse(fs.readFileSync(`${repo}/package-lock.json`, 'utf8'))
const result = {
  status: 'pending', date: '2026-09-07',
  commit: execFileSync('git', ['rev-parse', 'HEAD'], { cwd: repo, encoding: 'utf8' }).trim(),
  harness: { playwright: lock.packages['node_modules/playwright'].version,
    chromium: null, viewport: { width: 1400, height: 900 }, panelWidth: 320, url },
  fixture: { path: fixture, sha256: createHash('sha256').update(raw).digest('hex') },
  eventSource: 'engagementPicture(window.__store.getState().scenario).report.events',
  events: null, captures: [], pageErrors: null, consoleErrors: null,
}
let browser
let ctx
try {
  browser = await chromium.launch({ headless: true })
  result.harness.chromium = browser.version()
  ctx = await browser.newContext({ viewport: result.harness.viewport })
  await ctx.addInitScript(() => localStorage.setItem('rangeboard.rightpanel.width', '320'))
  const page = await ctx.newPage()
  result.pageErrors = []
  result.consoleErrors = []
  page.on('pageerror', e => result.pageErrors.push(e.message))
  page.on('console', m => { if (m.type() === 'error') result.consoleErrors.push(m.text()) })
  await page.goto(url, { waitUntil: 'load' })
  await page.waitForFunction(() => window.__store && document.querySelector('.konvajs-content canvas'))
  await page.evaluate(async raw => {
    const { parseAutosave } = await import('/src/io/persist.ts')
    window.__store.getState().loadScenario(parseAutosave(raw))
  }, raw)
  assert.deepEqual(await page.evaluate(() => window.__store.getState().scenario.entities.map(e => e.id)),
    JSON.parse(raw).entities.map(e => e.id))
  await page.click('#board-phase-execute')
  await page.waitForSelector('.right-panel [data-widget-type]')
  assert.equal(await page.locator('.right-panel').evaluate(el => el.getBoundingClientRect().width), 320)
  result.events = await page.evaluate(async () => {
    const { engagementPicture } = await import('/src/analysis/impact.ts')
    return engagementPicture(window.__store.getState().scenario).report.events
  })
  assert.equal(result.events.length, 4, 'Expected four deterministic events')
  const firstEffect = Math.min(...result.events.filter(e => e.killDelta > 0).map(e => e.tEffectS))
  assert.ok(firstEffect > 30, 'T+30 must precede the first modeled effect')
  const threatId = await page.evaluate(() =>
    window.__store.getState().scenario.entities.find(e => e.kind === 'uas').id)
  result.selectedThreatId = threatId
  const textOf = selector => page.locator(selector).allTextContents()
  const attrsOf = selector => page.locator(selector).evaluateAll(nodes => nodes.map(el => ({
    tag: el.tagName, text: el.textContent, title: el.getAttribute('title'),
    ariaLabel: el.getAttribute('aria-label'), role: el.getAttribute('role'),
  })))
  async function capture(name) {
    await page.evaluate(() => {
      window.__store.getState().clearSelection()
      window.__store.getState().setRightTab('analysis')
    })
    await page.waitForSelector('.right-panel [data-widget-type]')
    const strip = page.locator('.center > .engstrip')
    if (await strip.count() && !(await strip.locator('.engstrip-plot').count())) {
      await strip.locator('.engstrip-head').click()
    }
    await page.waitForTimeout(200)
    const capture = {
      name,
      playback: await page.evaluate(() => window.__store.getState().playback),
      rightPanelText: await textOf('.right-panel'),
      widgets: await page.locator('.right-panel [data-widget-type]').evaluateAll(nodes =>
        nodes.map(el => ({ type: el.dataset.widgetType, text: el.textContent }))),
      kpiTiles: await textOf('[data-widget-type="kpi-tiles"] .tile'),
      eventStripText: await textOf('.engstrip'),
      metadata: await attrsOf('.right-panel [title], .right-panel [aria-label], .engstrip [title]'),
      eventStripTooltips: [],
    }
    for (const plot of await page.locator('.engstrip-plot:visible').all()) {
      const box = await plot.boundingBox()
      if (!box) continue
      await page.mouse.move(box.x + Math.min(30, box.width / 2), box.y + 8)
      await page.waitForTimeout(100)
      capture.eventStripTooltips.push(...await textOf('.engstrip-tip'))
    }
    await page.evaluate(id => window.__store.getState().select({ entityIds: [id] }), threatId)
    await page.waitForSelector('.entity-workspace')
    await page.click('#entity-section-telemetry')
    await page.waitForTimeout(200)
    capture.selectedThreat = {
      compactText: await textOf('.readout.compact'),
      telemetryText: await textOf('#entity-workspace-content'),
      metadata: await attrsOf('#entity-workspace-content [title], .readout [title]'),
      trackTelemetryLayer: await page.evaluate(() => window.__store.getState().ui.layers.trackTelemetry),
    }
    // Overview preserves selection, so the screenshot can show widgets and compact telemetry.
    await page.locator('.entity-workspace-header button').click()
    await page.waitForSelector('.right-panel [data-widget-type]')
    await page.mouse.move(10, 10)
    await page.waitForTimeout(200)
    await page.screenshot({ path: path.join(outDir, `${name}.png`), fullPage: true })
    capture.screenshot = `${name}.png`
    result.captures.push(capture)
  }
  await capture('before-play')
  // Actual 1x playback reaches 30 s. Snap the paused cursor to exactly 30 s via real actions;
  // retain and report the actual observed horizon, including publication-step overshoot.
  await page.evaluate(() => {
    let reached = false
    const unsub = window.__store.subscribe(st => {
      if (reached || st.playback.simTimeS < 30) return
      reached = true
      unsub()
      window.__store.getState().pauseRun()
      window.__store.getState().scrubTo(30)
      window.__store.getState().pauseRun()
    })
    window.__store.getState().setSpeedMult(1)
    window.__store.getState().startRun()
  })
  await page.waitForFunction(() => {
    const p = window.__store.getState().playback
    return p.phase === 'paused' && p.simTimeS === 30
  }, null, { timeout: 45000 })
  assert.ok(await page.evaluate(() => window.__store.getState().playback.observedThroughS) < firstEffect)
  await capture('paused-t30')
  const forwardS = Math.max(...result.events.map(e => e.tEffectS)) + 1
  await page.evaluate(t => window.__store.getState().scrubTo(t), forwardS)
  await capture('scrub-forward')
  await page.evaluate(() => window.__store.getState().scrubTo(30))
  await capture('scrub-back')
  assert.deepEqual(result.pageErrors, [])
  assert.deepEqual(result.consoleErrors, [])
  result.status = 'captured'
} catch (error) {
  result.status = result.harness.chromium == null ? 'blocked-before-browser' : 'failed'
  result.error = String(error)
  process.exitCode = 1
} finally {
  await ctx?.close()
  await browser?.close()
  fs.writeFileSync(path.join(outDir, 'browser-baseline.json'), JSON.stringify(result, null, 2) + '\n')
}
