  const root = process.env.SERVE_DIR
  const types = { '.js': 'text/javascript', '.css': 'text/css', '.html': 'text/html', '.woff2': 'font/woff2', '.svg': 'image/svg+xml', '.map': 'application/json', '.json': 'application/json' }
  server = http.createServer((req, res) => {
    let file = path.join(root, decodeURIComponent(new URL(req.url, 'http://x').pathname))
    if (!fs.existsSync(file) || fs.statSync(file).isDirectory()) file = path.join(root, 'index.html')
    res.writeHead(200, { 'content-type': types[path.extname(file)] || 'application/octet-stream', 'cache-control': 'no-store' })
    fs.createReadStream(file).pipe(res)
  })
  await new Promise((r) => server.listen(Number(process.env.SERVE_PORT || 5392), '127.0.0.1', r))
}

async function metrics() {
  const { metrics } = await cdp.send('Performance.getMetrics')
  const m = Object.fromEntries(metrics.map((x) => [x.name, x.value]))
  return m
}

function aggregate(profile) {
  const byId = new Map(profile.nodes.map((n) => [n.id, n]))
  const self = new Map()
  const { samples, timeDeltas } = profile
  let total = 0
  for (let i = 0; i < samples.length; i++) {
    const dt = (timeDeltas[i] || 0) / 1000
    total += dt
    self.set(samples[i], (self.get(samples[i]) || 0) + dt)
  }
  const fn = new Map(), file = new Map(), special = new Map()
  for (const [id, ms] of self) {
    const n = byId.get(id)
    const cf = n.callFrame
    const short = cf.url ? cf.url.replace(/^https?:\/\/[^/]+/, '').replace(/\?.*$/, '') : ''
    if (!cf.url) { special.set(cf.functionName || '(anonymous)', (special.get(cf.functionName || '(anonymous)') || 0) + ms); continue }
    const fk = `${cf.functionName || '(anonymous)'} ${short}:${cf.lineNumber + 1}`
    fn.set(fk, (fn.get(fk) || 0) + ms)
    const fileKey = short.includes('node_modules') ? short.replace(/.*node_modules\/(\.vite\/deps\/)?/, 'npm:').replace(/\.js.*$/, '') : short
    file.set(fileKey, (file.get(fileKey) || 0) + ms)
  }
  const top = (m, n) => [...m].sort((a, b) => b[1] - a[1]).slice(0, n).map(([k, v]) => [k, Math.round(v)])
  return { totalMs: Math.round(total), special: top(special, 6), files: top(file, 18), fns: top(fn, 25) }
}

async function phase(name, fn) {
  await page.evaluate(() => {
    const s = { long: [], frames: [], sets: 0, scen: 0 }
    const po = new PerformanceObserver((l) => { for (const e of l.getEntries()) s.long.push(Math.round(e.duration)) })
    po.observe({ type: 'longtask' })
    let last = performance.now(); let raf = requestAnimationFrame(function tick(t) { s.frames.push(t - last); last = t; raf = requestAnimationFrame(tick) })
    const unsub = window.__store ? window.__store.subscribe((a, b) => { s.sets++; if (a.scenario !== b.scenario) s.scen++ }) : () => {}
    window.__sampler = { stop() { po.disconnect(); cancelAnimationFrame(raf); unsub(); const f = s.frames.slice(1); const sum = (x) => x.reduce((p, c) => p + c, 0)
      return { ms: Math.round(sum(f)), frames: f.length, fps: f.length ? +(f.length / (sum(f) / 1000)).toFixed(1) : null, over33: f.filter((x) => x > 33).length, over100: f.filter((x) => x > 100).length, maxFrame: f.length ? Math.round(Math.max(...f)) : null, longDurationsMs: s.long, longTasks: s.long.length, longMs: sum(s.long), longMax: s.long.length ? Math.max(...s.long) : 0, sets: s.sets, scenarioChanges: s.scen } } }
  })
  const m0 = await metrics()
  await cdp.send('Profiler.start')
  const t = Date.now()
  await fn()
  const wall = Date.now() - t
  const { profile } = await cdp.send('Profiler.stop')
  const m1 = await metrics()
  const inPage = await page.evaluate(() => window.__sampler.stop())
  const d = (k) => +(m1[k] - m0[k]).toFixed(3)
  const out = { phase: name, wallMs: wall, ...inPage,
    scriptMs: Math.round(d('ScriptDuration') * 1000), layoutMs: Math.round(d('LayoutDuration') * 1000), styleMs: Math.round(d('RecalcStyleDuration') * 1000), taskMs: Math.round(d('TaskDuration') * 1000),
    layouts: d('LayoutCount'), styleRecalcs: d('RecalcStyleCount'), domNodes: m1.Nodes, heapMB: +(m1.JSHeapUsedSize / 1048576).toFixed(1) }
  console.log(JSON.stringify(out))
  const agg = aggregate(profile)
  console.log(`  profile total ${agg.totalMs} ms; special: ${agg.special.map(([k, v]) => `${k}=${v}`).join(' ')}`)
  console.log('  files: ' + agg.files.map(([k, v]) => `${v}ms ${k}`).join(' | '))
  console.log('  fns:\n    ' + agg.fns.map(([k, v]) => `${v}ms ${k}`).join('\n    '))
}
