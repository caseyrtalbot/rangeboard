// Row G kill-types capture: 1265 x 900, the layered acoustic MEROPS laydown
// (src/playback/__fixtures__/layered-acoustic-merops.json, physics switches kept, ROE
// free, no doctrine) on the dev server. It starts sampled trials in Execute until one
// trial holds a pre-arrival hard kill, a pre-arrival mission kill and a late kill,
// completes the run, then reads and shoots the Now screen, the board readout pills, the
// run timeline, the Outcome screen, the Debrief threat record and the threat detail.
// Usage: node engine-g-capture.mjs <base-url> <out-dir> <repo-root> [doctrine-off]
// The fixture has no doctrine record, so parseAutosave fills the default (enabled). The
// optional fourth argument turns the fire doctrine off before the first trial.

// Per threat, the death under the one dead rule (deadTimeOf) and the planned arrival.
const trialInfo = () =>
  page.evaluate(async () => {
    const { deadTimeOf } = await import('/src/playback/deadTimes.ts')
    const s = window.__store.getState()
    const t = s.trial
    if (t.status !== 'ready') return { status: t.status }
    const rep = t.result.report
    const name = (id) => s.scenario.entities.find((e) => e.id === id)?.name ?? id
    const threats = rep.threats.map((row) => {
      const a = t.result.assessments.find((x) => x.uasId === row.uasId)
      const d = deadTimeOf(rep, row.uasId)
      return {
        uasId: row.uasId,
        name: name(row.uasId),
        shipCount: row.shipCount,
        arrivalS: a?.arrivalS ?? null,
        death: d,
        kind: !d ? 'alive' : d.late ? 'late' : d.mixed ? 'mixed' : d.soft ? 'mission' : 'hard',
      }
    })
    return {
      status: t.status,
      runId: t.manifest.runId,
      seed: t.manifest.seed,
      replicateIndex: t.manifest.replicateIndex,
      engineRevision: t.manifest.engineRevision,
      sweepStepS: t.manifest.sweepStepS,
      events: rep.events.length,
      ledger: {
        softShotKills: rep.events.filter((e) => e.softShot && e.killDelta === 1).length,
        lateEffectKills: rep.events.filter((e) => e.lateEffect && e.killDelta === 1).length,
        killDelta1: rep.events.filter((e) => e.killDelta === 1).length,
        flightCapped: rep.events.filter((e) => e.flightCapped).length,
      },
      threats,
    }
  })

async function scrub(t, why) {
  const before = await clock()

/** Board track picture, threat phase and run timeline rows of one threat at a horizon. */
const readAt = (h, uasId) =>
  page.evaluate(
    async ({ h, uasId }) => {
      const { latestEstimatesAt } = await import('/src/playback/trackEstimateView.ts')
      const { runTimelineOf } = await import('/src/analysis/runTimeline.ts')
      const { trialResultOf } = await import('/src/analysis/trialResult.ts')
      const { trialTelemetryAt } = await import('/src/panels/analysis/threatReadout.ts')
      const s = window.__store.getState()
      const t = s.trial
      const stepS = t.manifest.sweepStepS
      const latest = latestEstimatesAt(t.result.tracks, h).filter((e) => e.uasId === uasId)
      const drawn = latest.filter((e) => h - e.state.tS <= stepS + 1e-9)
      const view = trialResultOf(t.scenario, t.manifest, t.result)
      const tel = trialTelemetryAt(view, uasId, h)
      const model = runTimelineOf(view, h)
      const rows = []
      for (const g of model.groups)
        for (const r of g.rows) {
          if (r.entityId === uasId) rows.push(r)
          for (const sub of r.subLanes ?? []) if (sub.entityId === uasId) rows.push(sub)
        }
      const end = (list) => (list.length ? Math.max(...list.map((x) => x.endS)) : null)
      return {
        horizonS: h,
        observedThroughS: s.playback.observedThroughS,
        phase: tel.phase,
        members: tel.members.map((m) => m.outcome),
        latestOfThreat: latest.map((e) => ({ id: e.id, tS: e.state.tS })),
        drawnOfThreat: drawn.length,
        rows: rows.map((r) => ({
          key: r.key,
          lastSpanEndS: end(r.spans),
          holdEndS: end(r.spans.filter((x) => x.kind === 'hold' || x.kind === 'coast')),
          ghost: r.spans.filter((x) => x.kind === 'ghost').map((x) => [x.startS, x.endS]),
          killMarks: r.marks.filter((m) => ['hard-kill', 'mission-kill', 'drop', 'arrival'].includes(m.kind)).map((m) => [m.kind, m.tS, m.capped ? 'capped' : undefined].filter((x) => x != null)),
        })),
      }
    },
    { h, uasId },
  )

/** Scroll the nearest scrolling ancestor so the element sits at its top; the page never moves. */
const panelScrollTo = (selector) =>
  page.evaluate((sel) => {
    const target = document.querySelector(sel)
    if (!target) return false
    let el = target.parentElement
    while (el && !(['auto', 'scroll'].includes(getComputedStyle(el).overflowY) && el.scrollHeight > el.clientHeight)) el = el.parentElement
    if (el) el.scrollTop += target.getBoundingClientRect().top - el.getBoundingClientRect().top
    return !!el
  }, selector)

  await page.evaluate(

  )

/**
 * Show only the subject's rows in the strip: the sensors that held it (one opened on its
 * sub-lanes, the sensor whose hold of the subject ends last), its track row and its
 * threat row. View state only; the strip model is unchanged.
 */
async function focusStrip(uasId) {
  const kept = await page.evaluate(async (uasId) => {
    const { runTimelineOf } = await import('/src/analysis/runTimeline.ts')
    const { trialResultOf } = await import('/src/analysis/trialResult.ts')
    const s = window.__store.getState()
    const t = s.trial
    const model = runTimelineOf(trialResultOf(t.scenario, t.manifest, t.result), Infinity)
    s.showAllRunTimelineRows()
    const kept = []
    let open = null
    let openEnd = -1
    for (const g of model.groups)
      for (const r of g.rows) {
        const sub = (r.subLanes ?? []).find((x) => x.entityId === uasId)
        if (r.entityId === uasId || sub) {
          kept.push(r.key)
          const end = sub ? Math.max(-1, ...sub.spans.map((x) => x.endS)) : -1
          if (sub && end > openEnd) {
            open = r.key
            openEnd = end
          }
        } else s.hideRunTimelineRow(r.key)
      }
    s.expandAllRunTimeline(['sensor', 'track', 'threat'], [])
    window.__store.setState((st) => ({
      ui: { ...st.ui, runTimeline: { ...st.ui.runTimeline, openSensorsAll: false, openSensors: open ? [open] : [] } },
    }))
    return { kept, openSensor: open }
  }, uasId)
  await sleep(300)
  return kept
}

async function detail(subject, name) {
  await page.evaluate((id) => window.__store.getState().select({ entityIds: [id] }), subject.uasId)
  await waitFor(page, () => document.querySelector('#entity-section-telemetry'))

  await waitFor(page, () => document.querySelector('section[data-section="survival"]'))
  await panelScrollTo('section[data-section="survival"]')
  await sleep(400)
  const read = await page.evaluate(() => {
    const sec = document.querySelector('section[data-section="survival"]')
    const texts = [...sec.querySelectorAll('svg text')].map((t) => t.textContent.trim()).filter(Boolean)
    return { labels: texts.filter((x) => /^(FX|ARR|CPA)$/.test(x)), text: sec.textContent.replace(/\s+/g, ' ').trim().slice(0, 300) }
  })
  if (name) await shot(name, `Analyze, ${subject.name} selected, threat detail survival chart`)
  return read
}
