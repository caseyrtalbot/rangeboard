import assert from 'node:assert/strict'
import fs from 'node:fs'
import path from 'node:path'
import { createHash } from 'node:crypto'
import { execFileSync } from 'node:child_process'
import { chromium } from '/Users/caseytalbot/Projects/rangeboard/node_modules/playwright/index.mjs'

// Run with node --input-type=module - fixture url outDir repo < r2-capture.mjs.txt
// The target repo must serve its own source. Use an isolated browser and an owned port.
const [,, fixture, url, outDir, repo] = process.argv
assert.ok(fixture && url && outDir && repo, 'usage: fixture url outDir repo')
assert.notEqual(new URL(url).port, '5174', 'never the user dev server')
fs.mkdirSync(outDir, { recursive: true })
const raw = fs.readFileSync(fixture, 'utf8')
const result = {
  status: 'pending', date: '2026-09-07',
  commit: execFileSync('git', ['rev-parse', 'HEAD'], { cwd: repo, encoding: 'utf8' }).trim(),
  fixture: { path: fixture, sha256: createHash('sha256').update(raw).digest('hex') },
  url, viewport: { width: 1400, height: 900 }, panelWidth: 320,
  captures: [], pageErrors: [], consoleErrors: [],
}
let browser, ctx
try {
  browser = await chromium.launch({ headless: true })
  result.chromium = browser.version()
  ctx = await browser.newContext({ viewport: result.viewport })
  await ctx.addInitScript(() => localStorage.setItem('rangeboard.rightpanel.width', '320'))
  const page = await ctx.newPage()
  page.on('pageerror', e => result.pageErrors.push(e.message))
  page.on('console', m => { if (m.type() === 'error') result.consoleErrors.push(m.text()) })
  await page.goto(url, { waitUntil: 'load' })
  await page.waitForFunction(() => window.__store && document.querySelector('.konvajs-content canvas'))
  await page.evaluate(async raw => {
    const { parseAutosave } = await import('/src/io/persist.ts')
    window.__store.getState().loadScenario(parseAutosave(raw))
  }, raw)
  result.ids = await page.evaluate(() => {
    const entities = window.__store.getState().scenario.entities
    return { threat: entities.find(e => e.kind === 'uas').id,
      asset: entities.find(e => e.kind === 'asset')?.id ?? null,
      entities: entities.map(e => ({ id: e.id, name: e.name, kind: e.kind })) }
  })
  assert.deepEqual(result.ids.entities.map(e => e.id), JSON.parse(raw).entities.map(e => e.id))
  result.events = await page.evaluate(async () => {
    const { engagementPicture } = await import('/src/analysis/impact.ts')
    return engagementPicture(window.__store.getState().scenario).report.events
  })
  assert.ok(result.events.length > 0, 'fixture must have modeled effects')
  result.eventEffectTimesS = result.events.map(e => e.tEffectS)
  result.forwardS = Math.max(...result.eventEffectTimesS) + 5
  const firstEffectS = Math.min(...result.events.filter(e => e.killDelta > 0).map(e => e.tEffectS))
  assert.ok(firstEffectS > 30, 'paused cursor must precede first effect')
  result.layersKey = await page.evaluate(async () => (await import('/src/io/storageKeys.ts')).LAYERS_KEY)
  const textOf = sel => page.locator(sel).allTextContents()
  const attrsOf = sel => page.locator(sel).evaluateAll(nodes => nodes.map(el => ({
    tag: el.tagName, text: el.textContent, title: el.getAttribute('title'), ariaLabel: el.getAttribute('aria-label'),
  })))
  async function state() {
    return page.evaluate(key => {
      const s = window.__store.getState()
      return { playback: s.playback, boardPhase: s.ui.boardPhase ?? null,
        domBoardPhase: document.querySelector('[data-board-phase]')?.getAttribute('data-board-phase') ?? null,
        activePhaseButton: document.querySelector('.board-phases [aria-selected="true"]')?.id ?? null,
        layers: s.ui.layers, layersStorage: localStorage.getItem(key), selection: s.selection,
        rightTab: s.ui.rightTab, analysisLayout: s.ui.analysis.layout }
    }, result.layersKey)
  }
  async function phase(name) {
    await page.click(`#board-phase-${name}`)
    await page.waitForFunction(name => document.querySelector(`#board-phase-${name}`)?.getAttribute('aria-selected') === 'true', name)
    await page.waitForTimeout(250)
  }
  async function noSelection() {
    await page.evaluate(() => {
      window.__store.getState().clearSelection()
      window.__store.getState().setRightTab('analysis')
    })
  }
  async function threatSelection() {
    await page.evaluate(id => window.__store.getState().select({ entityIds: [id] }), result.ids.threat)
    await page.waitForTimeout(200)
    const telemetry = page.locator('#entity-section-telemetry')
    if (await telemetry.isVisible()) await telemetry.click()
  }
  async function openStrip() {
    const strip = page.locator('.center > .engstrip')
    if (await strip.count() && !(await strip.locator('.engstrip-plot').count())) {
      const head = strip.locator('.engstrip-head')
      if (await head.isVisible()) await head.click()
    }
  }
  async function shot(name, extra = {}, hover = false) {
    await page.mouse.move(10, 10)
    await openStrip()
    await page.waitForTimeout(350)
    const hoverRecord = { attempted: hover, target: null, tooltipText: [], attributes: [] }
    if (hover) {
      const plot = page.locator('.engstrip-plot:visible').first()
      if (await plot.count()) {
        const mark = plot.locator('svg > g').first().locator('rect, path').first()
        if (await mark.count()) {
          await mark.hover({ force: true })
          hoverRecord.target = await mark.evaluate(el => ({ tag: el.tagName, markup: el.outerHTML }))
        } else {
          const box = await plot.boundingBox()
          await page.mouse.move(box.x + Math.min(30, box.width / 2), box.y + 8)
          hoverRecord.target = { fallback: 'first lane plot at y+8' }
        }
        await page.waitForTimeout(150)
        hoverRecord.tooltipText = await textOf('.engstrip-tip, .engstrip [role="tooltip"]')
        hoverRecord.attributes = await attrsOf('.engstrip [title], .engstrip [aria-label]')
      }
    }
    const capture = { name, ...await state(),
      rightPanelText: await textOf('.right-panel'),
      engagementStripText: await textOf('.engstrip'),
      rightPanelAttributes: await attrsOf('.right-panel [title], .right-panel [aria-label]'),
      engagementStripAttributes: await attrsOf('.engstrip [title], .engstrip [aria-label]'),
      rightPanelTimeElements: await page.locator('.right-panel, .right-panel *').evaluateAll(nodes =>
        nodes.filter(el => /T\+\s*\d/.test(el.textContent ?? '')).map(el => ({
          tag: el.tagName, class: el.getAttribute('class'), text: el.textContent,
        }))),
      widgets: await page.locator('.right-panel [data-widget-type]').evaluateAll(nodes =>
        nodes.map(el => ({ type: el.dataset.widgetType, text: el.textContent }))),
      charts: await page.locator('.right-panel svg').evaluateAll(nodes => nodes.map(el => ({
        text: el.textContent, ariaLabel: el.getAttribute('aria-label'),
        ticks: [...el.querySelectorAll('.recharts-cartesian-axis-tick')].map(t => t.textContent),
      }))),
      hover: hoverRecord, screenshot: `${name}.png`, ...extra }
    await page.screenshot({ path: path.join(outDir, capture.screenshot), fullPage: true })
    result.captures.push(capture)
    fs.writeFileSync(path.join(outDir, 'r2-capture.json'), JSON.stringify(result, null, 2) + '\n')
    console.log(`captured ${name}`)
  }

  await phase('execute')
  await noSelection()
  await page.waitForSelector('.right-panel [data-widget-type]')
  assert.equal(await page.locator('.right-panel').evaluate(el => el.getBoundingClientRect().width), 320)
  await shot('execute-before-play')
  await threatSelection()
  await shot('execute-before-play-threat')
  await noSelection()
  // R0 flow: real 1x playback, pause at first publication >=30, snap back to 30.
  await page.evaluate(() => {
    let reached = false
    const unsub = window.__store.subscribe(st => {
      if (reached || st.playback.simTimeS < 30) return
      reached = true
      unsub()
      window.__store.getState().pauseRun()
      window.__store.getState().scrubTo(30)
      window.__store.getState().pauseRun()
    })
    window.__store.getState().setSpeedMult(1)
    window.__store.getState().startRun()
  })
  await page.waitForFunction(() => {
    const p = window.__store.getState().playback
    return p.phase === 'paused' && p.simTimeS === 30
  }, null, { timeout: 45000 })
  assert.ok((await state()).playback.observedThroughS < firstEffectS)
  await shot('execute-paused-t30', {}, true)
  await threatSelection()
  await shot('execute-paused-t30-threat')
  await page.evaluate(t => window.__store.getState().setSimTime(t), result.forwardS)
  await shot('execute-scrub-forward')
  await page.evaluate(() => window.__store.getState().setSimTime(10))
  await shot('execute-scrub-back')

  await page.evaluate(t => window.__store.getState().setSimTime(t), result.forwardS)
  const beforeSave = await state()
  const frame = await page.evaluate(() => {
    const s = window.__store.getState()
    const oldIds = new Set(s.scenario.frames.map(f => f.id))
    // The baseline calls this action saveFrame. Accommodate an after-tree rename.
    const action = typeof s.addFrame === 'function' ? 'addFrame' : 'saveFrame'
    s[action]('R2 forward cursor')
    const created = window.__store.getState().scenario.frames.find(f => !oldIds.has(f.id))
    if (!created) throw new Error('frame action did not create a frame')
    return { action, frame: created, playbackAfterSave: window.__store.getState().playback }
  })
  await page.evaluate(() => window.__store.getState().setSimTime(10))
  const beforeApply = await state()
  await page.evaluate(id => window.__store.getState().applyFrame(id), frame.frame.id)
  await shot('execute-frame-nav', { frameNavigation: { beforeSave, ...frame, beforeApply } })
  await noSelection()
  await phase('analyze')
  await shot('analyze')
  await phase('execute')
  await shot('execute-after-analyze')

  await page.evaluate(() => {
    window.__store.getState().setSimTime(30)
    window.__store.getState().pauseRun()
  })
  const previousLayout = (await state()).analysisLayout
  const widgetTypes = await page.evaluate(async () => {
    const { WIDGET_TYPES } = await import('/src/state/analysisLayout.ts')
    const s = window.__store.getState()
    for (const w of [...s.ui.analysis.layout.widgets]) s.removeAnalysisWidget(w.id)
    for (const type of WIDGET_TYPES) s.addAnalysisWidget(type)
    return [...WIDGET_TYPES]
  })
  assert.deepEqual((await state()).analysisLayout.widgets.map(w => w.type), widgetTypes)
  await page.waitForTimeout(400)
  await page.locator('details').evaluateAll(nodes => nodes.forEach(el => { el.open = true }))
  await shot('execute-custom-layout', { widgetTypes, previousLayout })
  await page.evaluate(id => window.__store.getState().applyLens(id), previousLayout.presetId)
  result.restoredLens = (await state()).analysisLayout.presetId

  const roundtrip = []
  for (const name of ['prepare', 'execute', 'analyze', 'prepare']) {
    await phase(name)
    roundtrip.push({ step: name, ...await state() })
  }
  await phase('execute')
  roundtrip.push({ step: 'execute-before-clean', ...await state() })
  await page.evaluate(() => window.__store.getState().applyLayerPreset('clean'))
  roundtrip.push({ step: 'execute-after-clean', ...await state() })
  await phase('prepare')
  roundtrip.push({ step: 'prepare-after-clean', ...await state() })
  await shot('layers-roundtrip', { roundtrip })
  assert.equal(result.captures.length, 11)
  assert.deepEqual(result.pageErrors, [], 'page errors fail capture')
  result.status = 'ok'
} catch (error) {
  result.status = 'failed'
  result.error = String(error)
  process.exitCode = 1
} finally {
  await ctx?.close()
  await browser?.close()
  if (result.pageErrors.length) { result.status = 'failed'; process.exitCode = 1 }
  fs.writeFileSync(path.join(outDir, 'r2-capture.json'), JSON.stringify(result, null, 2) + '\n')
}
